/* =========================================================
   VIEWER CONTAINER
========================================================= */

function waitForViewerContainer(
  element: HTMLElement,
  signal: AbortSignal
): Promise<void> {
  return new Promise(
    (
      resolve,
      reject
    ) => {
      let observer:
        ResizeObserver | null =
          null;

      let frameOne = 0;
      let frameTwo = 0;

      let finished =
        false;

      const cleanup =
        () => {
          observer?.disconnect();

          cancelAnimationFrame(
            frameOne
          );

          cancelAnimationFrame(
            frameTwo
          );

          signal.removeEventListener(
            "abort",
            handleAbort
          );
        };

      const handleAbort =
        () => {
          if (
            finished
          ) {
            return;
          }

          finished =
            true;

          cleanup();

          reject(
            new DOMException(
              "Viewer startup aborted.",
              "AbortError"
            )
          );
        };

      const check =
        () => {
          if (
            finished ||
            signal.aborted
          ) {
            return;
          }

          const rect =
            element.getBoundingClientRect();

          if (
            !element.isConnected ||
            rect.width <= 0 ||
            rect.height <= 0
          ) {
            return;
          }

          frameOne =
            requestAnimationFrame(
              () => {
                frameTwo =
                  requestAnimationFrame(
                    () => {
                      if (
                        finished ||
                        signal.aborted
                      ) {
                        return;
                      }

                      const finalRect =
                        element.getBoundingClientRect();

                      if (
                        finalRect.width > 0 &&
                        finalRect.height > 0
                      ) {
                        finished =
                          true;

                        cleanup();

                        resolve();
                      }
                    }
                  );
              }
            );
        };

      observer =
        new ResizeObserver(
          check
        );

      observer.observe(
        element
      );

      signal.addEventListener(
        "abort",
        handleAbort,
        {
          once: true,
        }
      );

      check();
    }
  );
}

function nextFrame():
  Promise<void> {
  return new Promise(
    (
      resolve
    ) => {
      requestAnimationFrame(
        () => {
          resolve();
        }
      );
    }
  );
}

/* =========================================================
   UNIT CONVERSION
========================================================= */

const UNIT_TO_MM: Record<
  EngineeringUnit,
  number
> = {
  inch: 25.4,
  mm: 1,
  cm: 10,
};

function parseFraction(
  value: string
):
  number | null {
  const clean =
    value.trim();

  const mixed =
    clean.match(
      /^(\d+)\s+(\d+)\/(\d+)$/
    );

  if (
    mixed
  ) {
    const whole =
      Number(
        mixed[1]
      );

    const numerator =
      Number(
        mixed[2]
      );

    const denominator =
      Number(
        mixed[3]
      );

    if (
      denominator ===
      0
    ) {
      return null;
    }

    return (
      whole +
      numerator /
        denominator
    );
  }

  const fraction =
    clean.match(
      /^(\d+)\/(\d+)$/
    );

  if (
    fraction
  ) {
    const numerator =
      Number(
        fraction[1]
      );

    const denominator =
      Number(
        fraction[2]
      );

    if (
      denominator ===
      0
    ) {
      return null;
    }

    return (
      numerator /
      denominator
    );
  }

  const parsed =
    Number(
      clean
    );

  return Number.isFinite(
    parsed
  )
    ? parsed
    : null;
}

function convertValue(
  value: number,
  from: EngineeringUnit,
  to: EngineeringUnit
) {
  return (
    value *
    UNIT_TO_MM[from] /
    UNIT_TO_MM[to]
  );
}

function formatConvertedValue(
  value: number
) {
  return value
    .toFixed(4)
    .replace(
      /\.?0+$/,
      ""
    );
}

function targetUnitLabel(
  unit:
    EngineeringUnit
) {
  return unit ===
    "inch"
    ? "in"
    : unit;
}

function explicitUnitToInternal(
  unit:
    string |
    undefined
):
  EngineeringUnit | null {
  if (
    !unit
  ) {
    return null;
  }

  const normalized =
    unit
      .trim()
      .toLowerCase();

  if (
    normalized === "\"" ||
    normalized === "â€³" ||
    normalized === "in" ||
    normalized === "inch" ||
    normalized === "inches"
  ) {
    return "inch";
  }

  if (
    normalized ===
    "mm"
  ) {
    return "mm";
  }

  if (
    normalized ===
    "cm"
  ) {
    return "cm";
  }

  return null;
}

/* =========================================================
   DIMENSION DETECTION
========================================================= */

function detectDimensionsInLine(
  lineText: string,
  pageIndex: number,
  lineIndex: number,
  sourceUnit:
    EngineeringUnit,
  targetUnit:
    EngineeringUnit,
  lineBox:
    PdfRectData
):
  DimensionCandidate[] {
  const results:
    DimensionCandidate[] =
      [];

  const text =
    lineText
      .replace(
        /\r?\n/g,
        " "
      )
      .trim();

  if (
    !text
  ) {
    return results;
  }

  const threadContext =
    /\b(?:UNC|UNF|UNEF|NPT|BSP|BSPT|TPI|THREAD)\b/i;

  const imperialThread =
    /\d+\s*\/\s*\d+\s*-\s*\d+/;

  if (
    threadContext.test(
      text
    ) ||
    imperialThread.test(
      text
    )
  ) {
    return results;
  }

  const metadataContext =
    /\b(?:PART\s*(?:NO|NUMBER)|DRAWING\s*(?:NO|NUMBER)|DWG|REV|REVISION|QTY|QUANTITY|DATE|SHEET|PAGE|SCALE|ANGLE|MATERIAL|WEIGHT|ITEM|PO\s*NUMBER)\b/i;

  const metadataLine =
    metadataContext.test(
      text
    );

  const dimensionPattern =
    /(Ã˜|âŒ€|R)?\s*((?:\d+\s+\d+\/\d+)|(?:\d+\/\d+)|(?:\d*\.\d+)|(?:\d+))\s*(inches?|inch|in|mm|cm|"|â€³)?(?:\s*Â±\s*((?:\d+\/\d+)|(?:\d*\.\d+)|(?:\d+)))?/gi;

  let match:
    RegExpExecArray |
    null;

  while (
    (
      match =
        dimensionPattern.exec(
          text
        )
    ) !== null
  ) {
    const prefix =
      match[1] ||
      "";

    const valueText =
      match[2];

    const explicitUnit =
      match[3];

    const tolerance =
      match[4];

    const original =
      match[0].trim();

    const after =
      text
        .slice(
          match.index +
            match[0].length
        )
        .trimStart()
        .charAt(
          0
        );

    if (
      after ===
      "Â°"
    ) {
      continue;
    }

    const hasPrefix =
      prefix.length >
      0;

    const hasFraction =
      valueText.includes(
        "/"
      );

    const hasDecimal =
      valueText.includes(
        "."
      );

    const hasTolerance =
      Boolean(
        tolerance
      );

    const hasUnit =
      Boolean(
        explicitUnit
      );

    if (
      !hasPrefix &&
      !hasFraction &&
      !hasDecimal &&
      !hasTolerance &&
      !hasUnit
    ) {
      continue;
    }

    if (
      metadataLine &&
      !hasPrefix &&
      !hasTolerance &&
      !hasUnit
    ) {
      continue;
    }

    const detectedUnit =
      explicitUnitToInternal(
        explicitUnit
      );

    if (
      detectedUnit &&
      detectedUnit !==
        sourceUnit
    ) {
      continue;
    }

    const numeric =
      parseFraction(
        valueText
      );

    if (
      numeric ===
      null
    ) {
      continue;
    }

    const convertedMain =
      formatConvertedValue(
        convertValue(
          numeric,
          sourceUnit,
          targetUnit
        )
      );

    let converted =
      `${prefix}${convertedMain}`;

    if (
      tolerance
    ) {
      const toleranceValue =
        parseFraction(
          tolerance
        );

      if (
        toleranceValue !==
        null
      ) {
        converted +=
          ` Â±${formatConvertedValue(
            convertValue(
              toleranceValue,
              sourceUnit,
              targetUnit
            )
          )}`;
      }
    }

    converted +=
      ` ${targetUnitLabel(
        targetUnit
      )}`;

    const confidence:
      DimensionConfidence =
        (
          hasPrefix ||
          hasUnit ||
          hasFraction ||
          hasTolerance
        )
          ? "high"
          : "review";

    results.push({
      id:
        `${pageIndex}-${lineIndex}-${match.index}-${results.length}`,

      pageIndex,

      pageNumber:
        pageIndex +
        1,

      original,

      converted,

      lineText:
        text,

      confidence,

      lineBox,
    });
  }

  return results;
}

async function getImageDimensions(
  file:
    File
):
  Promise<{
    width: number;
    height: number;
  }> {
  try {
    const bitmap =
      await createImageBitmap(
        file
      );

    const result = {
      width:
        bitmap.width,

      height:
        bitmap.height,
    };

    bitmap.close();

    return result;
  } catch {
    return {
      width: 400,
      height: 300,
    };
  }
}

      /* =====================================================
         LOCKED IMAGE HIT TESTING
      ===================================================== */

      function installImageHitTesting(
        instance:
          NutrientInstance,

        NutrientViewer:
          NutrientViewerType
      ) {
        imageHitTestCleanupRef.current?.();

        const contentDocument =
          instance.contentDocument;

        const handlePointerDown =
          (
            event:
              Event
          ) => {
            if (
              !contentEditorWorkflowRef.current ||

            }

            if (

            const pointerEvent =

            if (

              0

            }

            const target =
              pointerEvent.target as
                Element |
                null;

            if (
              target &&
              typeof target.closest ===
                "function" &&
              target.closest(
                [
                  "button",
                  "input",

                  "textarea",
                  "a",

                  "[contenteditable='true']",

                  ","
                )

            }

            const currentPage =
              instance
                .viewState
                .currentPageIndex;

            const candidatePages =
              [
                currentPage,
                currentPage -
                  1,
                currentPage +
                  1,
              ].filter(
                (
                  page,
                  index,
                  pages
                ) =>
                  page >=
                    0 &&
                  page <
                    instance
                      .totalPageCount &&
                  pages.indexOf(
                    page
                  ) ===
                    index
              );

            let imageId:
              string |
              null =
                null;

            for (
              const pageIndex
              of candidatePages
            ) {
              let point:
                any;

              try {
                point =
                  instance.transformContentClientToPageSpace(
                    new NutrientViewer
                      .Geometry
                      .Point({
                        x:
                          pointerEvent
                            .clientX,

                        y:
                          pointerEvent
                            .clientY,
                      }),

                    pageIndex
                  );
              } catch {
                continue;
              }

              const images =
                Array.from(
                  imageAnnotationsRef
                    .current
                    .values()
                )
                  .filter(
                    (
                      annotation:
                        any
                    ) =>
                      annotation
                        .pageIndex ===
                      pageIndex
                  )
                  .reverse();

              for (
                const image
                of images
              ) {
                const box =
                  image
                    .boundingBox;

                if (
                  !box
                ) {
                  continue;
                }

                const insideX =
                  point.x >=
                    box.left &&
                  point.x <=
                    box.left +
                      box.width;

                const insideY =
                  point.y >=
                    box.top &&
                  point.y <=
                    box.top +
                      box.height;

                if (
                  insideX &&
                  insideY
                ) {
                  if (
                    typeof image.id ===
                    "string"
                  ) {
                    imageId =
                      image.id;
                  }

                  break;
                }
              }

              if (
                imageId
              ) {
                break;
              }
            }

            if (
              !imageId
            ) {
              return;
            }

            pointerEvent.preventDefault();

            pointerEvent.stopPropagation();

            pointerEvent.stopImmediatePropagation();

            void selectImageForEditing(
              imageId
            );
          };

        contentDocument.addEventListener(
          "pointerdown",
          handlePointerDown,
          true
        );

        imageHitTestCleanupRef.current =
          () => {
            contentDocument.removeEventListener(
              "pointerdown",
              handlePointerDown,
              true
            );
          };
      }

      function rectCenter(
        rect:
          any
      ) {
        return {
          x:
            Number(
              rect?.left ??
              0
            ) +
            Number(
              rect?.width ??
              0
            ) /
              2,

          y:
            Number(
              rect?.top ??
              0
            ) +
            Number(
              rect?.height ??
              0
            ) /
              2,
        };
      }

      function distanceBetweenRects(
        first:
          any,

        second:
          any
      ) {
        if (
          !first ||
          !second
        ) {
          return Number
            .POSITIVE_INFINITY;
        }

        const firstCenter =
          rectCenter(
            first
          );

        const secondCenter =
          rectCenter(
            second
          );

        return (
          Math.abs(
            firstCenter.x -
              secondCenter.x
          ) +
          Math.abs(
            firstCenter.y -
              secondCenter.y
          )
        );
      }

      async function waitForMatchingContentBlock(
        pageIndex:
          number,

        term:
          string,

        resultRect:
          any
      ):
        Promise<any | null> {
        const instance =
          instanceRef.current;

        if (
          !instance
        ) {
          return null;
        }

        const searchTerm =
          term
            .trim()
            .toLocaleLowerCase();

        for (
          let attempt = 0;
          attempt <
            30;
          attempt++
        ) {
          const blocks =
            instance
              .contentEditor
              .getBlocks(
                pageIndex
              ) as
              readonly any[];

          if (
            blocks.length >
            0
          ) {
            const textBlocks =
              blocks.filter(
                (
                  block:
                    any
                ) =>
                  block &&
                  typeof block.id ===
                    "string" &&
                  typeof block.text ===
                    "string"
              );

            const matchingBlocks =
              textBlocks.filter(
                (
                  block:
                    any
                ) =>
                  block.text
                    .toLocaleLowerCase()
                    .includes(
                      searchTerm
                    )
              );

            const candidates =
              matchingBlocks.length >
              0
                ? matchingBlocks
                : textBlocks;

            if (
              candidates.length >
              0
            ) {
              let bestBlock =
                candidates[0];

              let bestDistance =
                Number
                  .POSITIVE_INFINITY;

              for (
                const block
                of candidates
              ) {
                const blockRect =
                  block
                    ?.boundingBox;

                if (
                  !blockRect
                ) {
                  if (
                    bestDistance ===
                    Number
                      .POSITIVE_INFINITY
                  ) {
                    bestBlock =
                      block;
                  }

                  continue;
                }

                const distance =
                  distanceBetweenRects(
                    blockRect,
                    resultRect
                  );

                if (
                  distance <
                  bestDistance
                ) {
                  bestDistance =
                    distance;

                  bestBlock =
                    block;
                }
              }

              return bestBlock;
            }
          }

          await delay(
            80
          );
        }

        return null;
      }

      async function openSearchResultInContentEditor(
        searchResult:
          any,

        term:
          string
      ) {
        const instance =
          instanceRef.current;

        const NutrientViewer =
          nutrientRef.current;

        if (
          !instance ||
          !NutrientViewer ||
          searchResultOpeningRef.current
        ) {
          return;
        }

        const cleanedTerm =
          term.trim();

        if (
          !cleanedTerm
        ) {
          return;
        }

        searchResultOpeningRef.current =
          true;

        try {
          const pageIndex =
            Number(
              searchResult
                ?.pageIndex
            );

          if (
            !Number.isInteger(
              pageIndex
            ) ||
            pageIndex <
              0 ||
            pageIndex >=
              instance
                .totalPageCount
          ) {
            return;
          }

          const rects =
            searchResult
              ?.rectsOnPage;

          if (
            !rects ||
            rects.size ===
              0
          ) {
            return;
          }

          const resultRect =
            NutrientViewer
              .Geometry
              .Rect
              .union(
                rects
              );

          /*
           * Search was started from the Content Editor workflow.
           * Keep that workflow active.
           */
          contentEditorWorkflowRef.current =
            true;

          resumeAfterSearchRef.current =
            false;

          /*
           * Stay at the searched result.
           * Do NOT zoom.
           */
          instance.jumpToRect(
            pageIndex,
            resultRect.grow(
              22
            )
          );

          /*
           * Search mode and Content Editor mode cannot
           * operate simultaneously.
           *
           * We now switch from Search to Content Editor.
           */
          activateContentEditor();

          /*
           * Allow page + blue Content Editor blocks
           * to render/detect.
           */
          await nextFrame();
          await nextFrame();

          const block =
            await waitForMatchingContentBlock(
              pageIndex,
              cleanedTerm,
              resultRect
            );

          if (
            !block ||
            typeof block.id !==
              "string"
          ) {
            /*
             * Content Editor is still open.
             * User can manually click the nearby blue box.
             */
            return;
          }

          /*
           * Put searched block directly into edit mode.
           */
          instance
            .contentEditor
            .focusBlock(
              block.id
            );

          await nextFrame();

          const text =
            typeof block.text ===
            "string"
              ? block.text
              : "";

          const lowerText =
            text.toLocaleLowerCase();

          const lowerTerm =
            cleanedTerm
              .toLocaleLowerCase();

          const matchIndex =
            lowerText.indexOf(
              lowerTerm
            );

          if (
            matchIndex <
            0
          ) {
            /*
             * Block is focused even when exact character
             * mapping is unavailable.
             */
            return;
          }

          /*
           * Select only the searched word inside
           * the blue editable text block.
           */
          await instance
            .contentEditor
            .setTextSelection(
              block.id,
              {
                begin:
                  matchIndex,

                end:
                  matchIndex +
                  cleanedTerm.length,
              }
            );
        } catch (
          error
        ) {
          console.error(
            "Unable to open searched text in Content Editor:",
            error
          );
        } finally {
          searchResultOpeningRef.current =
            false;
        }
      }

      function installSearchToEditBehavior(
        instance:
          NutrientInstance,

        NutrientViewer:
          NutrientViewerType
      ) {
        searchEditCleanupRef.current?.();

        const contentDocument =
          instance
            .contentDocument;

        /*
         * Keep the latest built-in search state.
         */
        const handleSearchStateChange =
          (
            state:
              any
          ) => {
            searchStateRef.current =
              state;
          };

        instance.addEventListener(
          "search.stateChange",
          handleSearchStateChange as any
        );

        searchStateRef.current =
          instance
            .searchState;

        /*
         * User searches normally.
         *
         * Yellow search result remains visible.
         *
         * Clicking that highlighted search result
         * moves it directly into Content Editor.
         */
        const handleSearchResultPointerDown =
          (
            event:
              Event
          ) => {
            const pointerEvent =
              event as
                PointerEvent;

            if (
              pointerEvent.button !==
              0
            ) {
              return;
            }

            if (
              searchResultOpeningRef.current
            ) {
              return;
            }

            if (
              instance
                .viewState
                .interactionMode !==
              NutrientViewer
                .InteractionMode
                .SEARCH
            ) {
              return;
            }

            const searchState =
              searchStateRef.current ??
              instance.searchState;

            const term =
              String(
                searchState
                  ?.term ??
                ""
              ).trim();

            const results =
              searchState
                ?.results;

            if (
              !term ||
              !results ||
              results.size ===
                0
            ) {
              return;
            }

            /*
             * Don't steal Search UI button/input clicks.
             */
            const target =
              pointerEvent.target as
                Element |
                null;

            if (
              target &&
              typeof target.closest ===
                "function" &&
              target.closest(
                [
                  "button",
                  "input",
                  "select",
                  "textarea",
                  "a",
                  "[role='button']",
                ].join(
                  ","
                )
              )
            ) {
              return;
            }

            let hitResult:
              any =
                null;

            results.forEach(
              (
                result:
                  any
              ) => {
                if (
                  hitResult
                ) {
                  return;
                }

                const pageIndex =
                  Number(
                    result
                      ?.pageIndex
                  );

                const rects =
                  result
                    ?.rectsOnPage;

                if (
                  !Number.isInteger(
                    pageIndex
                  ) ||
                  !rects ||
                  rects.size ===
                    0
                ) {
                  return;
                }

                let point:
                  any;

                try {
                  point =
                    instance.transformContentClientToPageSpace(
                      new NutrientViewer
                        .Geometry
                        .Point({
                          x:
                            pointerEvent
                              .clientX,

                          y:
                            pointerEvent
                              .clientY,
                        }),

                      pageIndex
                    );
                } catch {
                  return;
                }

                rects.forEach(
                  (
                    rect:
                      any
                  ) => {
                    if (
                      hitResult
                    ) {
                      return;
                    }

                    /*
                     * Small padding makes clicking the
                     * yellow highlight easier.
                     */
                    const padding =
                      5;

                    const left =
                      rect.left -
                      padding;

                    const top =
                      rect.top -
                      padding;

                    const right =
                      rect.left +
                      rect.width +
                      padding;

                    const bottom =
                      rect.top +
                      rect.height +
                      padding;

                    if (
                      point.x >=
                        left &&
                      point.x <=
                        right &&
                      point.y >=
                        top &&
                      point.y <=
                        bottom
                    ) {
                      hitResult =
                        result;
                    }
                  }
                );
              }
            );

            if (
              !hitResult
            ) {
              return;
            }

            pointerEvent.preventDefault();

            pointerEvent.stopPropagation();

            pointerEvent.stopImmediatePropagation();

            void openSearchResultInContentEditor(
              hitResult,
              term
            );
          };

        contentDocument.addEventListener(
          "pointerdown",
          handleSearchResultPointerDown,
          true
        );

        searchEditCleanupRef.current =
          () => {
            contentDocument.removeEventListener(
              "pointerdown",
              handleSearchResultPointerDown,
              true
            );

            try {
              instance.removeEventListener(
                "search.stateChange",
                handleSearchStateChange as any
              );
            } catch {
              // Already removed.
            }
          };
      }

      async function insertImage() {
        const instance =
          instanceRef.current;

        const NutrientViewer =
          nutrientRef.current;

        if (
          !instance ||
          !NutrientViewer ||
          imageInsertBusyRef.current
        ) {
          return;
        }

        const input =
          document.createElement(
            "input"
          );

        input.type =
          "file";

        input.accept =
          "image/png,image/jpeg,image/jpg";

        input.style.display =
          "none";

        document.body.appendChild(
          input
        );

        input.onchange =
          async () => {
            const file =
              input.files?.[0];

            input.remove();

            if (
              !file
            ) {
              return;
            }

            imageInsertBusyRef.current =
              true;

            try {
              await finishContentEditorIfNeeded();

              const imageAttachmentId =
                await instance.createAttachment(
                  file
                );

              const dimensions =
                await getImageDimensions(
                  file
                );

              const pageIndex =
                instance
                  .viewState
                  .currentPageIndex;

              const pageInfo =
                instance.pageInfoForIndex(
                  pageIndex
                );

              if (
                !pageInfo
              ) {
                throw new Error(
                  "Unable to read current PDF page size."
                );
              }

              const maxWidth =
                pageInfo.width *
                0.35;

              const maxHeight =
                pageInfo.height *
                0.35;

              const ratio =
                dimensions.width /
                Math.max(
                  dimensions.height,
                  1
                );

              let width =
                Math.min(
                  maxWidth,
                  Math.max(
                    120,
                    dimensions.width *
                      0.5
                  )
                );

              let height =
                width /
                ratio;

              if (
                height >
                maxHeight
              ) {
                height =
                  maxHeight;

                width =
                  height *
                  ratio;
              }

              width =
                Math.max(
                  80,
                  width
                );

              height =
                Math.max(
                  60,
                  height
                );

              const left =
                Math.max(
                  10,
                  (
                    pageInfo.width -
                    width
                  ) /
                    2
                );

              const top =
                Math.max(
                  10,
                  (
                    pageInfo.height -
                    height
                  ) /
                    2
                );

              const annotationId =
                NutrientViewer
                  .generateInstantId();

              const annotation =
                new NutrientViewer
                  .Annotations
                  .ImageAnnotation({
                    id:
                      annotationId,

                    pageIndex,

                    contentType:
                      file.type ||
                      "image/png",

                    imageAttachmentId,

                    fileName:
                      file.name,

                    boundingBox:
                      new NutrientViewer
                        .Geometry
                        .Rect({
                          left,
                          top,
                          width,
                          height,
                        }),

                    locked:
                      false,

                    lockedContents:
                      false,

                    readOnly:
                      false,
                  });

              await instance.create(
                annotation
              );

              cacheImageAnnotation(
                annotation
              );

              /*
               * LOCKED WORKING BEHAVIOR:
               * inserted image immediately gets handles.
               */
              await selectImageForEditing(
                annotationId
              );
            } catch (
              error
            ) {
              console.error(
                "Image insertion error:",
                error
              );

              alert(
                error instanceof
                  Error
                  ? error.message
                  : "Unable to insert image."
              );

              imageEditingRef.current =
                false;

              imageSelectionConfirmedRef.current =
                false;

              if (
                contentEditorWorkflowRef.current
              ) {
                activateContentEditor();
              }
            } finally {
              imageInsertBusyRef.current =
                false;
            }
          };

        input.oncancel =
          () => {
            input.remove();
          };

        input.click();
      }

      /* =====================================================
         CANCEL -> UNDO
      ===================================================== */

      function installUndoButton(
        instance:
          NutrientInstance
      ) {
        undoCleanupRef.current?.();

        const contentDocument =
          instance.contentDocument;

        let frame =
          0;

        const patch =
          () => {
            cancelAnimationFrame(
              frame
            );

            frame =
              requestAnimationFrame(
                () => {
                  if (
                    !instance
                      .contentEditor
                      .isActive()
                  ) {
                    return;
                  }

                  const buttons =
                    Array.from(
                      contentDocument
                        .querySelectorAll(
                          "button"
                        )
                    ) as HTMLButtonElement[];

                  const existing =
                    buttons.find(
                      (
                        button
                      ) =>
                        button
                          .dataset
                          .pdfUndo ===
                        "true"
                    );

                  const active =
                    instance
                      .contentEditor
                      .getActiveBlock();

                  const selected =
                    instance
                      .contentEditor
                      .getSelectedBlock();

                  const enabled =
                    Boolean(
                      active ||
                      selected
                    );

                  if (
                    existing
                  ) {
                    existing.disabled =
                      !enabled;

                    existing.style.opacity =
                      enabled
                        ? "1"
                        : "0.45";

                    existing.style.cursor =
                      enabled
                        ? "pointer"
                        : "not-allowed";

                    return;
                  }

                  const cancel =
                    buttons.find(
                      (
                        button
                      ) => {
                        const text =
                          button
                            .textContent
                            ?.trim()
                            .toLowerCase();

                        const aria =
                          button
                            .getAttribute(
                              "aria-label"
                            )
                            ?.trim()
                            .toLowerCase();

                        const title =
                          button
                            .getAttribute(
                              "title"
                            )
                            ?.trim()
                            .toLowerCase();

                        return (
                          text ===
                            "cancel" ||
                          aria ===
                            "cancel" ||
                          title ===
                            "cancel"
                        );
                      }
                    );

                  if (
                    !cancel
                  ) {
                    return;
                  }

                  const undo =
                    cancel.cloneNode(
                      true
                    ) as HTMLButtonElement;

                  undo.dataset.pdfUndo =
                    "true";

                  undo.textContent =
                    "Undo";

                  undo.setAttribute(
                    "title",
                    "Undo"
                  );

                  undo.setAttribute(
                    "aria-label",
                    "Undo"
                  );

                  undo.disabled =
                    !enabled;

                  undo.addEventListener(
                    "click",
                    (
                      event
                    ) => {
                      event.preventDefault();

                      event.stopPropagation();

                      event.stopImmediatePropagation();

                      try {
                        let block =
                          instance
                            .contentEditor
                            .getActiveBlock();

                        if (
                          !block
                        ) {
                          block =
                            instance
                              .contentEditor
                              .getSelectedBlock();

                          if (
                            block
                          ) {
                            instance
                              .contentEditor
                              .focusBlock(
                                block
                              );
                          }
                        }

                        requestAnimationFrame(
                          () => {
                            try {
                              if (
                                instance
                                  .contentEditor
                                  .canUndo()
                              ) {
                                instance
                                  .contentEditor
                                  .undo();
                              }
                            } catch (
                              error
                            ) {
                              console.error(
                                "Undo error:",
                                error
                              );
                            }

                            setTimeout(
                              patch,
                              0
                            );
                          }
                        );
                      } catch (
                        error
                      ) {
                        console.error(
                          "Undo error:",
                          error
                        );
                      }
                    },
                    true
                  );

                  cancel.replaceWith(
                    undo
                  );
                }
              );
          };

        const observer =
          new MutationObserver(
            patch
          );

        observer.observe(
          contentDocument,
          {
            childList:
              true,

            subtree:
              true,
          }
        );

        const stateChanged =
          () => {
            patch();
          };

        instance.addEventListener(
          "contentEditor.stateChange",
          stateChanged
        );

        patch();

        undoCleanupRef.current =
          () => {
            observer.disconnect();

            cancelAnimationFrame(
              frame
            );

            try {
              instance.removeEventListener(
                "contentEditor.stateChange",
                stateChanged
              );
            } catch {
              // Already removed.
            }
          };
      }

      /* =====================================================
         TEXT BOX Y-AXIS RESIZE

         IMPORTANT:
         This affects TextAnnotation only.

         Existing behaviour remains:
         - native left/right X resize
         - Content Editor
         - image re-selection
         - search
         - Unit Converter
         - save/download
         - pan/zoom

         Added behaviour:
         - top Y handle
         - bottom Y handle
         - drag upward/downward to resize height
      ===================================================== */

      function installVerticalTextBoxResize(
        instance:
          NutrientInstance,

        NutrientViewer:
          NutrientViewerType
      ) {
        verticalTextResizeCleanupRef.current?.();

        const TOP_HANDLE_ID =
          "pdf-text-y-resize-top";

        const BOTTOM_HANDLE_ID =
          "pdf-text-y-resize-bottom";

        const MIN_TEXT_HEIGHT =
          28;

        const PAGE_MARGIN =
          4;

        let selectedTextAnnotation:
          any =
            null;

        let topHandle:
          HTMLDivElement |
          null =
            null;

        let bottomHandle:
          HTMLDivElement |
          null =
            null;

        let activeDragCleanup:
          (() => void) |
          null =
            null;

        /* -----------------------------------------------
           REMOVE ONLY VISUAL HANDLES
        ----------------------------------------------- */

        function removeHandleOverlays() {
          try {
            instance.removeCustomOverlayItem(
              TOP_HANDLE_ID
            );
          } catch {
            // No top handle.
          }

          try {
            instance.removeCustomOverlayItem(
              BOTTOM_HANDLE_ID
            );
          } catch {
            // No bottom handle.
          }

          topHandle =
            null;

          bottomHandle =
            null;
        }

        /* -----------------------------------------------
           STOP ACTIVE Y DRAG
        ----------------------------------------------- */

        function stopActiveDrag() {
          activeDragCleanup?.();

          activeDragCleanup =
            null;
        }

        /* -----------------------------------------------
           CREATE HANDLE
        ----------------------------------------------- */

        function createHandle(
          edge:
            "top" |
            "bottom"
        ) {
          const handle =
            document.createElement(
              "div"
            );

          handle.style.width =
            "14px";

          handle.style.height =
            "14px";

          handle.style.borderRadius =
            "999px";

          handle.style.background =
            "#4f46e5";

          handle.style.border =
            "2px solid white";

          handle.style.boxShadow =
            "0 1px 4px rgba(0,0,0,0.28)";

          handle.style.boxSizing =
            "border-box";

          handle.style.cursor =
            "ns-resize";

          handle.style.pointerEvents =
            "auto";

          handle.style.touchAction =
            "none";

          handle.style.transform =
            "translate(-50%, -50%)";

          handle.style.zIndex =
            "999";

          handle.setAttribute(
            "title",
            edge ===
              "top"
              ? "Resize text box upward/downward"
              : "Resize text box downward/upward"
          );

          handle.setAttribute(
            "aria-label",
            edge ===
              "top"
              ? "Resize text box from top"
              : "Resize text box from bottom"
          );

          return handle;
        }

        /* -----------------------------------------------
           PLACE / UPDATE Y HANDLES
        ----------------------------------------------- */

        function updateHandlePositions(
          annotation:
            any
        ) {
          if (
            !annotation ||
            !annotation.boundingBox ||
            typeof annotation.pageIndex !==
              "number"
          ) {
            removeHandleOverlays();

            return;
          }

          const box =
            annotation.boundingBox;

          const centerX =
            box.left +
            box.width /
              2;

          if (
            !topHandle
          ) {
            topHandle =
              createHandle(
                "top"
              );

            topHandle.addEventListener(
              "pointerdown",
              (
                event
              ) => {
                startVerticalResize(
                  "top",
                  event,
                  annotation
                );
              }
            );
          }

          if (
            !bottomHandle
          ) {
            bottomHandle =
              createHandle(
                "bottom"
              );

            bottomHandle.addEventListener(
              "pointerdown",
              (
                event
              ) => {
                startVerticalResize(
                  "bottom",
                  event,
                  annotation
                );
              }
            );
          }

          instance.setCustomOverlayItem(
            new NutrientViewer
              .CustomOverlayItem({
                id:
                  TOP_HANDLE_ID,

                node:
                  topHandle,

                pageIndex:
                  annotation.pageIndex,

                position:
                  new NutrientViewer
                    .Geometry
                    .Point({
                      x:
                        centerX,

                      y:
                        box.top,
                    }),
              })
          );

          instance.setCustomOverlayItem(
            new NutrientViewer
              .CustomOverlayItem({
                id:
                  BOTTOM_HANDLE_ID,

                node:
                  bottomHandle,

                pageIndex:
                  annotation.pageIndex,

                position:
                  new NutrientViewer
                    .Geometry
                    .Point({
                      x:
                        centerX,

                      y:
                        box.top +
                        box.height,
                    }),
              })
          );
        }

        /* -----------------------------------------------
           START VERTICAL RESIZE
        ----------------------------------------------- */

        function startVerticalResize(
          edge:
            "top" |
            "bottom",

          pointerDownEvent:
            PointerEvent,

          startingAnnotation:
            any
        ) {
          if (
            !startingAnnotation ||
            !startingAnnotation.boundingBox ||
            typeof startingAnnotation.pageIndex !==
              "number"
          ) {
            return;
          }

          pointerDownEvent.preventDefault();

          pointerDownEvent.stopPropagation();

          pointerDownEvent.stopImmediatePropagation();

          stopActiveDrag();

          let currentAnnotation =
            startingAnnotation;

          const pageIndex =
            startingAnnotation.pageIndex;

          const originalBox =
            startingAnnotation.boundingBox;

          const fixedTop =
            originalBox.top;

          const fixedBottom =
            originalBox.top +
            originalBox.height;

          const pageInfo =
            instance.pageInfoForIndex(
              pageIndex
            );

          if (
            !pageInfo
          ) {
            return;
          }

          let animationFrame =
            0;

          let latestPointerEvent:
            PointerEvent |
            null =
              null;

          const viewerWindow =
            instance.contentWindow;

          /* -------------------------------------------
             APPLY RESIZE
          ------------------------------------------- */

          const applyResize =
            () => {
              animationFrame =
                0;

              const event =
                latestPointerEvent;

              latestPointerEvent =
                null;

              if (
                !event
              ) {
                return;
              }

              let point:
                any;

              try {
                point =
                  instance
                    .transformContentClientToPageSpace(
                      new NutrientViewer
                        .Geometry
                        .Point({
                          x:
                            event.clientX,

                          y:
                            event.clientY,
                        }),

                      pageIndex
                    );
              } catch {
                return;
              }

              const oldBox =
                currentAnnotation
                  .boundingBox;

              if (
                !oldBox
              ) {
                return;
              }

              let newTop =
                oldBox.top;

              let newHeight =
                oldBox.height;

              /* ---------------------------------------
                 DRAG BOTTOM HANDLE
              --------------------------------------- */

              if (
                edge ===
                "bottom"
              ) {
                const maximumBottom =
                  pageInfo.height -
                  PAGE_MARGIN;

                const newBottom =
                  Math.min(
                    maximumBottom,

                    Math.max(
                      fixedTop +
                        MIN_TEXT_HEIGHT,

                      point.y
                    )
                  );

                newTop =
                  fixedTop;

                newHeight =
                  newBottom -
                  fixedTop;
              }

              /* ---------------------------------------
                 DRAG TOP HANDLE
              --------------------------------------- */

              if (
                edge ===
                "top"
              ) {
                const minimumTop =
                  PAGE_MARGIN;

                const maximumTop =
                  fixedBottom -
                  MIN_TEXT_HEIGHT;

                newTop =
                  Math.max(
                    minimumTop,

                    Math.min(
                      maximumTop,
                      point.y
                    )
                  );

                newHeight =
                  fixedBottom -
                  newTop;
              }

              const newBoundingBox =
                new NutrientViewer
                  .Geometry
                  .Rect({
                    left:
                      oldBox.left,

                    top:
                      newTop,

                    width:
                      oldBox.width,

                    height:
                      newHeight,
                  });

              currentAnnotation =
                currentAnnotation
                  .set(
                    "boundingBox",
                    newBoundingBox
                  )
                  .set(
                    "isFitting",
                    false
                  )
                  .set(
                    "locked",
                    false
                  )
                  .set(
                    "lockedContents",
                    false
                  )
                  .set(
                    "readOnly",
                    false
                  );

              selectedTextAnnotation =
                currentAnnotation;

              /*
               * Update actual PDF annotation.
               */
              void instance
                .update(
                  currentAnnotation
                )
                .catch(
                  (
                    error:
                      unknown
                  ) => {
                    console.error(
                      "Text box Y resize error:",
                      error
                    );
                  }
                );

              /*
               * Move top/bottom handles together
               * with the resized bounding box.
               */
              updateHandlePositions(
                currentAnnotation
              );
            };

          /* -------------------------------------------
             POINTER MOVE
          ------------------------------------------- */

          const handlePointerMove =
            (
              event:
                PointerEvent
            ) => {
              latestPointerEvent =
                event;

              event.preventDefault();

              if (
                animationFrame ===
                0
              ) {
                animationFrame =
                  requestAnimationFrame(
                    applyResize
                  );
              }
            };

          /* -------------------------------------------
             POINTER UP
          ------------------------------------------- */

          const finishResize =
            () => {
              if (
                animationFrame !==
                0
              ) {
                cancelAnimationFrame(
                  animationFrame
                );

                animationFrame =
                  0;
              }

              viewerWindow.removeEventListener(
                "pointermove",
                handlePointerMove,
                true
              );

              viewerWindow.removeEventListener(
                "pointerup",
                finishResize,
                true
              );

              viewerWindow.removeEventListener(
                "pointercancel",
                finishResize,
                true
              );

              activeDragCleanup =
                null;

              if (
                selectedTextAnnotation &&
                typeof selectedTextAnnotation.id ===
                  "string"
              ) {
                try {
                  instance.setSelectedAnnotations(
                    NutrientViewer
                      .Immutable
                      .List([
                        selectedTextAnnotation.id,
                      ])
                  );
                } catch {
                  // Selection already exists.
                }
              }

              if (
                selectedTextAnnotation
              ) {
                updateHandlePositions(
                  selectedTextAnnotation
                );
              }
            };

          viewerWindow.addEventListener(
            "pointermove",
            handlePointerMove,
            true
          );

          viewerWindow.addEventListener(
            "pointerup",
            finishResize,
            true
          );

          viewerWindow.addEventListener(
            "pointercancel",
            finishResize,
            true
          );

          activeDragCleanup =
            () => {
              viewerWindow.removeEventListener(
                "pointermove",
                handlePointerMove,
                true
              );

              viewerWindow.removeEventListener(
                "pointerup",
                finishResize,
                true
              );

              viewerWindow.removeEventListener(
                "pointercancel",
                finishResize,
                true
              );

              if (
                animationFrame !==
                0
              ) {
                cancelAnimationFrame(
                  animationFrame
                );
              }
            };
        }

        /* -----------------------------------------------
           READ CURRENT TEXT SELECTION
        ----------------------------------------------- */

        function refreshSelectedTextAnnotation() {
          if (
            activeDragCleanup
          ) {
            return;
          }

          const selected =
            instance
              .getSelectedAnnotations();

          if (
            !selected ||
            selected.size !==
              1
          ) {
            selectedTextAnnotation =
              null;

            removeHandleOverlays();

            return;
          }

          const annotation =
            selected.get(
              0
            );

          if (
            !(
              annotation instanceof
              NutrientViewer
                .Annotations
                .TextAnnotation
            )
          ) {
            selectedTextAnnotation =
              null;

            removeHandleOverlays();

            return;
          }

          selectedTextAnnotation =
            annotation;

          updateHandlePositions(
            annotation
          );
        }

        /* -----------------------------------------------
           SELECTION CHANGE
        ----------------------------------------------- */

        const handleSelectionChange =
          () => {
            window.setTimeout(
              refreshSelectedTextAnnotation,
              0
            );
          };

        /* -----------------------------------------------
           TEXT ANNOTATION UPDATED
        ----------------------------------------------- */

        const handleAnnotationUpdate =
          (
            annotations:
              any
          ) => {
            if (
              !selectedTextAnnotation ||
              activeDragCleanup ||
              !annotations
            ) {
              return;
            }

            annotations.forEach(
              (
                annotation:
                  any
              ) => {
                if (
                  annotation?.id ===
                    selectedTextAnnotation?.id &&
                  annotation instanceof
                    NutrientViewer
                      .Annotations
                      .TextAnnotation
                ) {
                  selectedTextAnnotation =
                    annotation;

                  updateHandlePositions(
                    annotation
                  );
                }
              }
            );
          };

        /* -----------------------------------------------
           TEXT ANNOTATION DELETED
        ----------------------------------------------- */

        const handleAnnotationDelete =
          (
            annotations:
              any
          ) => {
            if (
              !selectedTextAnnotation ||
              !annotations
            ) {
              return;
            }

            let selectedWasDeleted =
              false;

            annotations.forEach(
              (
                annotation:
                  any
              ) => {
                if (
                  annotation?.id ===
                  selectedTextAnnotation?.id
                ) {
                  selectedWasDeleted =
                    true;
                }
              }
            );

            if (
              selectedWasDeleted
            ) {
              selectedTextAnnotation =
                null;

              stopActiveDrag();

              removeHandleOverlays();
            }
          };

        instance.addEventListener(
          "annotationSelection.change",
          handleSelectionChange
        );

        instance.addEventListener(
          "annotations.update",
          handleAnnotationUpdate
        );

        instance.addEventListener(
          "annotations.delete",
          handleAnnotationDelete
        );

        /*
         * In case a text annotation is already selected
         * when this feature is installed.
         */
        refreshSelectedTextAnnotation();

        verticalTextResizeCleanupRef.current =
          () => {
            stopActiveDrag();

            removeHandleOverlays();

            try {
              instance.removeEventListener(
                "annotationSelection.change",
                handleSelectionChange
              );

              instance.removeEventListener(
                "annotations.update",
                handleAnnotationUpdate
              );

              instance.removeEventListener(
                "annotations.delete",
                handleAnnotationDelete
              );
            } catch {
              // Already removed.
            }
          };
      }

      /* =====================================================
         MOUSE PAN + MOUSE WHEEL ZOOM
      ===================================================== */

      function installMouseControls(
        instance:
          NutrientInstance,

        NutrientViewer:
          NutrientViewerType,

        attempt =
          0
      ) {
        mouseCleanupRef.current?.();

        const contentDocument =
          instance.contentDocument;

        const contentWindow =
          instance.contentWindow;

        const scroll =
          contentDocument.querySelector(
            ".PSPDFKit-Scroll"
          ) as HTMLElement | null;

        if (
          !scroll
        ) {
          if (
            attempt <
            12
          ) {
            mouseInstallTimerRef.current =
              window.setTimeout(
                () => {
                  installMouseControls(
                    instance,
                    NutrientViewer,
                    attempt +
                      1
                  );
                },
                150
              );
          }

          return;
        }

        let dragging =
          false;

        let startX =
          0;

        let startY =
          0;

        let startLeft =
          0;

        let startTop =
          0;

        let wheelBusy =
          false;

        const isPanMode =
          () =>
            instance
              .viewState
              .interactionMode ===
            NutrientViewer
              .InteractionMode
              .PAN;

        const isInteractive =
          (
            target:
              EventTarget |
              null
          ) => {
            if (
              !target
            ) {
              return false;
            }

            const element =
              target as
                Element;

            if (
              typeof element.closest !==
              "function"
            ) {
              return false;
            }

            return Boolean(
              element.closest(
                [
                  "button",
                  "input",
                  "select",
                  "textarea",
                  "a",
                  "[role='button']",
                  "[contenteditable='true']",
                ].join(
                  ","
                )
              )
            );
          };

        const mouseDown =
          (
            event:
              MouseEvent
          ) => {
            /*
             * IMPORTANT:
             * never steal mouse events from selected images.
             */
            if (
              imageEditingRef.current
            ) {
              return;
            }

            if (
              event.button !==
                0 ||
              !isPanMode() ||
              isInteractive(
                event.target
              )
            ) {
              return;
            }

            dragging =
              true;

            startX =
              event.clientX;

            startY =
              event.clientY;

            startLeft =
              scroll.scrollLeft;

            startTop =
              scroll.scrollTop;

            scroll.style.cursor =
              "grabbing";

            scroll.style.userSelect =
              "none";

            clearLocateHighlight();

            event.preventDefault();
          };

        const mouseMove =
          (
            event:
              MouseEvent
          ) => {
            if (
              !dragging
            ) {
              return;
            }

            scroll.scrollLeft =
              startLeft -
              (
                event.clientX -
                startX
              );

            scroll.scrollTop =
              startTop -
              (
                event.clientY -
                startY
              );

            event.preventDefault();
          };

        const mouseUp =
          () => {
            if (
              !dragging
            ) {
              return;
            }

            dragging =
              false;

            scroll.style.cursor =
              isPanMode()
                ? "grab"
                : "";

            scroll.style.userSelect =
              "";
          };

        const wheel =
          (
            event:
              WheelEvent
          ) => {
            event.preventDefault();

            event.stopPropagation();

            if (
              wheelBusy
            ) {
              return;
            }

            wheelBusy =
              true;

            requestAnimationFrame(
              () => {
                try {
                  const current =
                    instance
                      .currentZoomLevel;

                  const min =
                    instance
                      .minimumZoomLevel;

                  const max =
                    instance
                      .maximumZoomLevel;

                  let next =
                    event.deltaY <
                    0
                      ? current *
                        1.12
                      : current /
                        1.12;

                  next =
                    Math.max(
                      min,
                      Math.min(
                        max,
                        next
                      )
                    );

                  instance.setViewState(
                    (
                      state
                    ) =>
                      state.set(
                        "zoom",
                        next
                      )
                  );
                } finally {
                  wheelBusy =
                    false;
                }
              }
            );
          };

        const viewChanged =
          () => {
            const currentMode =
              instance
                .viewState
                .interactionMode;

            const previousMode =
              previousInteractionModeRef.current;

            if (
              resumeAfterSearchRef.current &&
              previousMode ===
                NutrientViewer
                  .InteractionMode
                  .SEARCH &&
              currentMode !==
                NutrientViewer
                  .InteractionMode
                  .SEARCH
            ) {
              resumeAfterSearchRef.current =
                false;

              setTimeout(
                () => {
                  if (
                    contentEditorWorkflowRef.current &&
                    currentMode !==
                      NutrientViewer
                        .InteractionMode
                        .CONTENT_EDITOR
                  ) {
                    activateContentEditor();
                  }
                },
                80
              );
            }

            previousInteractionModeRef.current =
              currentMode;
          };

        previousInteractionModeRef.current =
          instance
            .viewState
            .interactionMode;

        scroll.addEventListener(
          "mousedown",
          mouseDown,
          true
        );

        contentWindow.addEventListener(
          "mousemove",
          mouseMove,
          true
        );

        contentWindow.addEventListener(
          "mouseup",
          mouseUp,
          true
        );

        contentWindow.addEventListener(
          "blur",
          mouseUp
        );

        scroll.addEventListener(
          "wheel",
          wheel,
          {
            passive:
              false,

            capture:
              true,
          }
        );

        instance.addEventListener(
          "viewState.change",
          viewChanged
        );

        mouseCleanupRef.current =
          () => {
            scroll.removeEventListener(
              "mousedown",
              mouseDown,
              true
            );

            contentWindow.removeEventListener(
              "mousemove",
              mouseMove,
              true
            );

            contentWindow.removeEventListener(
              "mouseup",
              mouseUp,
              true
            );

            contentWindow.removeEventListener(
              "blur",
              mouseUp
            );

            scroll.removeEventListener(
              "wheel",
              wheel,
              true
            );

            try {
              instance.removeEventListener(
                "viewState.change",
                viewChanged
              );
            } catch {
              // Already removed.
            }
          };
      }

      /* =====================================================
         EXPORT
      ===================================================== */

        Promise<ArrayBuffer> {
        const instance =
          instanceRef.current;

        if (
          !instance
        ) {
          throw new Error(
            "PDF editor is not ready."
          );
        }

        if (
          instance
            .contentEditor
            .isActive()
        ) {
          return await instance
            .exportContentEditorPDF();
        }

        return await instance.exportPDF();
      }

          async analyzeDimensions(
            options
          ) {
            const instance =
              instanceRef.current;

            if (
              !instance
            ) {
              throw new Error(
                "PDF editor is not ready."
              );
            }

            if (
              options.sourceUnit ===
              options.targetUnit
            ) {
              throw new Error(
                "Source and target units must be different."
              );
            }

            const totalPages =
              instance
                .totalPageCount;

            const currentPage =
              instance
                .viewState
                .currentPageIndex;

            const pages =
              options.scope ===
              "all"
                ? Array.from(
                    {
                      length:
                        totalPages,
                    },
                    (
                      _,
                      index
                    ) =>
                      index
                  )
                : [
                    currentPage,
                  ];

            const dimensions:
              DimensionCandidate[] =
                [];

            for (
              const pageIndex
              of pages
            ) {
              const lines =
                await instance
                  .textLinesForPageIndex(
                    pageIndex
                  );

              lines.forEach(
                (
                  line,
                  lineIndex
                ) => {
                  const box =
                    line
                      .boundingBox
                      .toJS();

                  dimensions.push(
                    ...detectDimensionsInLine(
                      line.contents,
                      pageIndex,
                      lineIndex,
                      options.sourceUnit,
                      options.targetUnit,
                      {
                        left:
                          box.left,

                        top:
                          box.top,

                        width:
                          box.width,

                        height:
                          box.height,
                      }
                    )
                  );
                }
              );
            }

            return {
              dimensions,

              pagesAnalyzed:
                pages.length,

              totalPages,
            };
          },

          async locateDimension(
            dimension
          ) {
            const instance =
              instanceRef.current;

            const NutrientViewer =
              nutrientRef.current;

            if (
              !instance ||
              !NutrientViewer
            ) {
              throw new Error(
                "PDF editor is not ready."
              );
            }

            clearLocateHighlight();

            let targetRect =
              new NutrientViewer
                .Geometry
                .Rect({
                  left:
                    dimension
                      .lineBox
                      .left,

                  top:
                    dimension
                      .lineBox
                      .top,

                  width:
                    dimension
                      .lineBox
                      .width,

                  height:
                    dimension
                      .lineBox
                      .height,
                });

            try {
              const results =
                await instance.search(
                  dimension.original,
                  {
                    startPageIndex:
                      dimension
                        .pageIndex,

                    endPageIndex:
                      dimension
                        .pageIndex,
                  }
                );

              const targetX =
                targetRect.left +
                targetRect.width /
                  2;

              const targetY =
                targetRect.top +
                targetRect.height /
                  2;

              let best =
                Number
                  .POSITIVE_INFINITY;

              results.forEach(
                (
                  result
                ) => {
                  if (
                    result.pageIndex !==
                      dimension
                        .pageIndex ||
                    result
                      .rectsOnPage
                      .size ===
                      0
                  ) {
                    return;
                  }

                  const candidate =
                    NutrientViewer
                      .Geometry
                      .Rect
                      .union(
                        result
                          .rectsOnPage
                      );

                  const x =
                    candidate.left +
                    candidate.width /
                      2;

                  const y =
                    candidate.top +
                    candidate.height /
                      2;

                  const distance =
                    Math.abs(
                      x -
                        targetX
                    ) +
                    Math.abs(
                      y -
                        targetY
                    ) *
                      3;

                  if (
                    distance <
                    best
                  ) {
                    best =
                      distance;

                    targetRect =
                      candidate;
                  }
                }
              );
            } catch {
              // Use original line rect.
            }

            instance.jumpToRect(
              dimension.pageIndex,
              targetRect.grow(
                18
              )
            );

            activatePanMode();

            const highlight =
              document.createElement(
                "div"
              );

            highlight.style.width =
              `${Math.max(
                targetRect.width,
                8
              )}px`;

            highlight.style.height =
              `${Math.max(
                targetRect.height,
                8
              )}px`;

            highlight.style.background =
              "rgba(37,99,235,0.28)";

            highlight.style.border =
              "2px solid #2563eb";

            highlight.style.borderRadius =
              "3px";

            highlight.style.boxSizing =
              "border-box";

            highlight.style.boxShadow =
              "0 0 0 3px rgba(37,99,235,0.16)";

            highlight.style.pointerEvents =
              "none";

            instance.setCustomOverlayItem(
              new NutrientViewer
                .CustomOverlayItem({
                  id:
                    LOCATE_OVERLAY_ID,

                  node:
                    highlight,

                  pageIndex:
                    dimension
                      .pageIndex,

                  position:
                    new NutrientViewer
                      .Geometry
                      .Point({
                        x:
                          targetRect.left,

                        y:
                          targetRect.top,
                      }),
                })
            );
          },

          async saveToBackend() {
            const instance =
              instanceRef.current;

            if (
              !instance
            ) {
              throw new Error(
                "PDF editor is not ready."
              );
            }

            const pdfBuffer =
              await exportWithoutClosingEditor();

            const blob =
              new Blob(
                [
                  pdfBuffer,
                ],
                {
                  type:
                    "application/pdf",
                }
              );

            const formData =
              new FormData();

            formData.append(
              "file",
              blob,
              `${documentId}.pdf`
            );

            const response =
              await fetch(
                `http://127.0.0.1:8000/api/documents/${documentId}/save`,
                {
                  method:
                    "POST",

                  body:
                    formData,
                }
              );

            if (
              !response.ok
            ) {
              let message =
                "Unable to save edited PDF.";

              try {
                const data =
                  await response.json();

                if (
                  typeof data
                    ?.detail ===
                  "string"
                ) {
                  message =
                    data.detail;
                }
              } catch {
                // Keep default.
              }

              throw new Error(
                message
              );
            }

            let warning:
              string |
              undefined;

            if (
              instance
                .contentEditor
                .isActive()
            ) {
              const result =
                await instance
                  .saveContentEditingSession();

              if (
                result.status ===
                  "reported" &&
                result
                  .uncoveredCodepoints
                  .length >
                  0
              ) {
                warning =
                  "Some edited characters may not be supported by the original PDF fonts.";
              }

              if (
                result.status ===
                "reportUnavailable"
              ) {
                warning =
                  "PDF saved, but font support could not be fully verified.";
              }
            }

            contentEditorWorkflowRef.current =
              false;

            imageEditingRef.current =
              false;

            imageSelectionConfirmedRef.current =
              false;

            searchResultOpeningRef.current =
              false;

            try {
              instance.setSelectedAnnotations(
                null
              );
            } catch {
              // Nothing selected.
            }

            activatePanMode();

            return {
              warning,
            };
          },

          async downloadPdf(
            filename
          ) {
            const pdfBuffer =
              await exportWithoutClosingEditor();

            const blob =
              new Blob(
                [
                  pdfBuffer,
                ],
                {
                  type:
                    "application/pdf",
                }
              );

            const url =
              URL.createObjectURL(
                blob
              );

            const cleanName =
              (
                filename ||
                "document.pdf"
              ).replace(
                /\.pdf$/i,
                ""
              );

            const link =
              document.createElement(
                "a"
              );

            link.href =
              url;

            link.download =
              `${cleanName}-edited.pdf`;

            document.body.appendChild(
              link
            );

            link.click();

            link.remove();

            setTimeout(
              () => {
                URL.revokeObjectURL(
                  url
                );
              },
              1000
            );

            return {};
          },
        })
      );

      /* =====================================================
         LOAD VIEWER
      ===================================================== */

      useEffect(
        () => {
          if (
            !documentId
          ) {
            return;
          }

          let cancelled =
            false;

          let loadedInstance:
            NutrientInstance |
            null =
              null;

          let selectionHandler:
            ((annotations: any) => void) |
            null =
              null;

          const controller =
            new AbortController();

          const load =
            async () => {
              const container =
                viewerRef.current;

              if (
                !container
              ) {
                return;
              }

              try {
                onLoadingChangeRef
                  .current?.(
                    true
                  );

                onReadyRef
                  .current?.(
                    false
                  );

                onErrorRef
                  .current?.(
                    ""
                  );

                const key =
                  process.env
                    .NEXT_PUBLIC_NUTRIENT_KEY;

                if (
                  !key
                ) {
                  throw new Error(
                    "NEXT_PUBLIC_NUTRIENT_KEY is missing."
                  );
                }

                const response =
                  await fetch(
                    `http://127.0.0.1:8000/api/documents/${documentId}/file`,
                    {
                      signal:
                        controller
                          .signal,
                    }
                  );

                if (
                  !response.ok
                ) {
                  throw new Error(
                    `Unable to load PDF. HTTP ${response.status}`
                  );
                }

                const buffer =
                  await response
                    .arrayBuffer();

                const module =
                  await import(
                    "@nutrient-sdk/viewer"
                  );

                const NutrientViewer =
                  module.default;

                nutrientRef.current =
                  NutrientViewer;

                await waitForViewerContainer(
                  container,
                  controller.signal
                );

                if (
                  cancelled
                ) {
                  return;
                }

                try {
                  NutrientViewer.unload(
                    container
                  );
                } catch {
                  // Nothing loaded.
                }

                const instance =
                  await NutrientViewer.load({
                    container,

                    baseUrl:
                      `${window.location.origin}/`,

                    session:
                      key,

                    document:
                      buffer,

                    initialViewState:
                      new NutrientViewer
                        .ViewState({
                          showToolbar:
                            true,

                          interactionMode:
                            NutrientViewer
                              .InteractionMode
                              .PAN,
                        }),

                    signal:
                      controller
                        .signal,
                  });

                if (
                  cancelled
                ) {
                  return;
                }

                loadedInstance =
                  instance;

                instanceRef.current =
                  instance;

                applyMainToolbar(
                  instance
                );

                removeNoteOption(
                  instance
                );

                installUndoButton(
                  instance
                );

                /*
                 * TextAnnotation only:
                 * adds top + bottom Y-axis resize handles.
                 */
                installVerticalTextBoxResize(
                  instance,
                  NutrientViewer
                );

                /*
                 * LOCKED image behavior.
                 */
                installImageAnnotationEvents(
                  instance,
                  NutrientViewer
                );

                installImageHitTesting(
                  instance,
                  NutrientViewer
                );

                /*
                 * NEW search -> editable blue box behavior.
                 */
                installSearchToEditBehavior(
                  instance,
                  NutrientViewer
                );

                await hydrateImageCacheForPage(
                  instance
                    .viewState
                    .currentPageIndex
                );

                activatePanMode();

                /*
                 * LOCKED image deselection behavior.
                 */
                selectionHandler =
                  (
                    annotations:
                      any
                  ) => {
                    if (
                      !imageEditingRef.current
                    ) {
                      return;
                    }

                    if (
                      annotations &&
                      annotations.size >
                        0
                    ) {
                      imageSelectionConfirmedRef.current =
                        true;

                      return;
                    }

                    if (
                      !imageSelectionConfirmedRef.current
                    ) {
                      return;
                    }

                    returnToContentEditorAfterImage();
                  };

                instance.addEventListener(
                  "annotationSelection.change",
                  selectionHandler
                );

                mouseInstallTimerRef.current =
                  window.setTimeout(
                    () => {
                      if (
                        !cancelled
                      ) {
                        installMouseControls(
                          instance,
                          NutrientViewer
                        );
                      }
                    },
                    500
                  );

                onReadyRef
                  .current?.(
                    true
                  );

                onLoadingChangeRef
                  .current?.(
                    false
                  );
              } catch (
                error
              ) {
                if (
                  cancelled ||
                  controller
                    .signal
                    .aborted
                ) {
                  return;
                }

                console.error(
                  "Nutrient load error:",
                  error
                );

                onErrorRef
                  .current?.(
                    error instanceof
                      Error
                      ? error.message
                      : "Unable to load PDF."
                  );

                onReadyRef
                  .current?.(
                    false
                  );

                onLoadingChangeRef
                  .current?.(
                    false
                  );
              }
            };

          void load();

          return () => {
            cancelled =
              true;

            controller.abort();

            if (
              mouseInstallTimerRef.current !==
              null
            ) {
              clearTimeout(
                mouseInstallTimerRef.current
              );

              mouseInstallTimerRef.current =
                null;
            }

            mouseCleanupRef.current?.();

            mouseCleanupRef.current =
              null;

            imageHitTestCleanupRef.current?.();

            imageHitTestCleanupRef.current =
              null;

            imageEventsCleanupRef.current?.();

            imageEventsCleanupRef.current =
              null;

            searchEditCleanupRef.current?.();

            searchEditCleanupRef.current =
              null;

            undoCleanupRef.current?.();

            undoCleanupRef.current =
              null;

            imageAnnotationsRef.current.clear();

            searchStateRef.current =
              null;

            imageEditingRef.current =
              false;

            imageSelectionConfirmedRef.current =
              false;

            contentEditorWorkflowRef.current =
              false;

            searchResultOpeningRef.current =
              false;

            resumeAfterSearchRef.current =
              false;

            if (
              loadedInstance &&
              selectionHandler
            ) {
              try {
                loadedInstance.removeEventListener(
                  "annotationSelection.change",
                  selectionHandler
                );
              } catch {
                // Already removed.
              }
            }

            if (
              loadedInstance
            ) {
              try {
                loadedInstance.removeCustomOverlayItem(
                  LOCATE_OVERLAY_ID
                );
              } catch {
                // Nothing.
              }
            }

            const container =
              viewerRef.current;

            const NutrientViewer =
              nutrientRef.current;

            if (
              container &&
              NutrientViewer
            ) {
              try {
                NutrientViewer.unload(
                  container
                );
              } catch {
                // Nothing.
              }
            }

            instanceRef.current =
              null;

            nutrientRef.current =
              null;
          };
        },
        [
          documentId,
        ]
      );

      return (
        <div
          ref={
            viewerRef
          }
          className="h-full w-full"
          style={{
            width:
              "100%",

            height:
              "100%",

            minWidth:
              "1px",

            minHeight:
              "420px",
          }}
        />
      );
    }
  );

const NutrientEditor =
  memo(
    NutrientEditorComponent
  );

NutrientEditor.displayName =
  "NutrientEditor";

export default NutrientEditor;
