/* =========================================================
   VIEWER CONTAINER
========================================================= */

function waitForViewerContainer(
  element: HTMLElement,
  signal: AbortSignal
): Promise<void> {
  return new Promise(
    (
      resolve,
      reject
    ) => {
      let observer:
        ResizeObserver | null =
          null;

      let frameOne = 0;
      let frameTwo = 0;

      let finished =
        false;

      const cleanup =
        () => {
          observer?.disconnect();

          cancelAnimationFrame(
            frameOne
          );

          cancelAnimationFrame(
            frameTwo
          );

          signal.removeEventListener(
            "abort",
            handleAbort
          );
        };

      const handleAbort =
        () => {
          if (
            finished
          ) {
            return;
          }

          finished =
            true;

          cleanup();

          reject(
            new DOMException(
              "Viewer startup aborted.",
              "AbortError"
            )
          );
        };

      const check =
        () => {
          if (
            finished ||
            signal.aborted
          ) {
            return;
          }

          const rect =
            element.getBoundingClientRect();

          if (
            !element.isConnected ||
            rect.width <= 0 ||
            rect.height <= 0
          ) {
            return;
          }

          frameOne =
            requestAnimationFrame(
              () => {
                frameTwo =
                  requestAnimationFrame(
                    () => {
                      if (
                        finished ||
                        signal.aborted
                      ) {
                        return;
                      }

                      const finalRect =
                        element.getBoundingClientRect();

                      if (
                        finalRect.width > 0 &&
                        finalRect.height > 0
                      ) {
                        finished =
                          true;

                        cleanup();

                        resolve();
                      }
                    }
                  );
              }
            );
        };

      observer =
        new ResizeObserver(
          check
        );

      observer.observe(
        element
      );

      signal.addEventListener(
        "abort",
        handleAbort,
        {
          once: true,
        }
      );

      check();
    }
  );
}

function nextFrame():
  Promise<void> {
  return new Promise(
    (
      resolve
    ) => {
      requestAnimationFrame(
        () => {
          resolve();
        }
      );
    }
  );
}

/* =========================================================
   UNIT CONVERSION
========================================================= */

const UNIT_TO_MM: Record<
  EngineeringUnit,
  number
> = {
  inch: 25.4,
  mm: 1,
  cm: 10,
};

function parseFraction(
  value: string
):
  number | null {
  const clean =
    value.trim();

  const mixed =
    clean.match(
      /^(\d+)\s+(\d+)\/(\d+)$/
    );

  if (
    mixed
  ) {
    const whole =
      Number(
        mixed[1]
      );

    const numerator =
      Number(
        mixed[2]
      );

    const denominator =
      Number(
        mixed[3]
      );

    if (
      denominator ===
      0
    ) {
      return null;
    }

    return (
      whole +
      numerator /
        denominator
    );
  }

  const fraction =
    clean.match(
      /^(\d+)\/(\d+)$/
    );

  if (
    fraction
  ) {
    const numerator =
      Number(
        fraction[1]
      );

    const denominator =
      Number(
        fraction[2]
      );

    if (
      denominator ===
      0
    ) {
      return null;
    }

    return (
      numerator /
      denominator
    );
  }

  const parsed =
    Number(
      clean
    );

  return Number.isFinite(
    parsed
  )
    ? parsed
    : null;
}

function convertValue(
  value: number,
  from: EngineeringUnit,
  to: EngineeringUnit
) {
  return (
    value *
    UNIT_TO_MM[from] /
    UNIT_TO_MM[to]
  );
}

function formatConvertedValue(
  value: number
) {
  return value
    .toFixed(4)
    .replace(
      /\.?0+$/,
      ""
    );
}

function targetUnitLabel(
  unit:
    EngineeringUnit
) {
  return unit ===
    "inch"
    ? "in"
    : unit;
}

function explicitUnitToInternal(
  unit:
    string |
    undefined
):
  EngineeringUnit | null {
  if (
    !unit
  ) {
    return null;
  }

  const normalized =
    unit
      .trim()
      .toLowerCase();

  if (
    normalized === "\"" ||
    normalized === "â€³" ||
    normalized === "in" ||
    normalized === "inch" ||
    normalized === "inches"
  ) {
    return "inch";
  }

  if (
    normalized ===
    "mm"
  ) {
    return "mm";
  }

  if (
    normalized ===
    "cm"
  ) {
    return "cm";
  }

  return null;
}

/* =========================================================
   DIMENSION DETECTION
========================================================= */

function detectDimensionsInLine(
  lineText: string,
  pageIndex: number,
  lineIndex: number,
  sourceUnit:
    EngineeringUnit,
  targetUnit:
    EngineeringUnit,
  lineBox:
    PdfRectData
):
  DimensionCandidate[] {
  const results:
    DimensionCandidate[] =
      [];

  const text =
    lineText
      .replace(
        /\r?\n/g,
        " "
      )
      .trim();

  if (
    !text
  ) {
    return results;
  }

  const threadContext =
    /\b(?:UNC|UNF|UNEF|NPT|BSP|BSPT|TPI|THREAD)\b/i;

  const imperialThread =
    /\d+\s*\/\s*\d+\s*-\s*\d+/;

  if (
    threadContext.test(
      text
    ) ||
    imperialThread.test(
      text
    )
  ) {
    return results;
  }

  const metadataContext =
    /\b(?:PART\s*(?:NO|NUMBER)|DRAWING\s*(?:NO|NUMBER)|DWG|REV|REVISION|QTY|QUANTITY|DATE|SHEET|PAGE|SCALE|ANGLE|MATERIAL|WEIGHT|ITEM|PO\s*NUMBER)\b/i;

  const metadataLine =
    metadataContext.test(
      text
    );

  const dimensionPattern =
    /(Ã˜|âŒ€|R)?\s*((?:\d+\s+\d+\/\d+)|(?:\d+\/\d+)|(?:\d*\.\d+)|(?:\d+))\s*(inches?|inch|in|mm|cm|"|â€³)?(?:\s*Â±\s*((?:\d+\/\d+)|(?:\d*\.\d+)|(?:\d+)))?/gi;

  let match:
    RegExpExecArray |
    null;

  while (
    (
      match =
        dimensionPattern.exec(
          text
        )
    ) !== null
  ) {
    const prefix =
      match[1] ||
      "";

    const valueText =
      match[2];

    const explicitUnit =
      match[3];

    const tolerance =
      match[4];

    const original =
      match[0].trim();

    const after =
      text
        .slice(
          match.index +
            match[0].length
        )
        .trimStart()
        .charAt(
          0
        );

    if (
      after ===
      "Â°"
    ) {
      continue;
    }

    const hasPrefix =
      prefix.length >
      0;

    const hasFraction =
      valueText.includes(
        "/"
      );

    const hasDecimal =
      valueText.includes(
        "."
      );

    const hasTolerance =
      Boolean(
        tolerance
      );

    const hasUnit =
      Boolean(
        explicitUnit
      );

    if (
      !hasPrefix &&
      !hasFraction &&
      !hasDecimal &&
      !hasTolerance &&
      !hasUnit
    ) {
      continue;
    }

    if (
      metadataLine &&
      !hasPrefix &&
      !hasTolerance &&
      !hasUnit
    ) {
      continue;
    }

    const detectedUnit =
      explicitUnitToInternal(
        explicitUnit
      );

    if (
      detectedUnit &&
      detectedUnit !==
        sourceUnit
    ) {
      continue;
    }

    const numeric =
      parseFraction(
        valueText
      );

    if (
      numeric ===
      null
    ) {
      continue;
    }

    const convertedMain =
      formatConvertedValue(
        convertValue(
          numeric,
          sourceUnit,
          targetUnit
        )
      );

    let converted =
      `${prefix}${convertedMain}`;

    if (
      tolerance
    ) {
      const toleranceValue =
        parseFraction(
          tolerance
        );

      if (
        toleranceValue !==
        null
      ) {
        converted +=
          ` Â±${formatConvertedValue(
            convertValue(
              toleranceValue,
              sourceUnit,
              targetUnit
            )
          )}`;
      }
    }

    converted +=
      ` ${targetUnitLabel(
        targetUnit
      )}`;

    const confidence:
      DimensionConfidence =
        (
          hasPrefix ||
          hasUnit ||
          hasFraction ||
          hasTolerance
        )
          ? "high"
          : "review";

    results.push({
      id:
        `${pageIndex}-${lineIndex}-${match.index}-${results.length}`,

      pageIndex,

      pageNumber:
        pageIndex +
        1,

      original,

      converted,

      lineText:
        text,

      confidence,

      lineBox,
    });
  }

  return results;
}

/* =========================================================

async function getImageDimensions(
  file:
    File
):
  Promise<{
    width: number;
    height: number;
  }> {
  try {
    const bitmap =
      await createImageBitmap(
        file
      );

    const result = {
      width:
        bitmap.width,

      height:
        bitmap.height,
    };

    bitmap.close();

    return result;
  } catch {
    return {
      width: 400,
      height: 300,
    };
  }
}

            let imageId:

            for (

              try {
                point =

                      .Geometry

                        x:

                        y:
                          pointerEvent
                            .clientY,
                      }),

                    pageIndex
                  );
              } catch {
                continue;
              }

              const images =
                Array.from(
                  imageAnnotationsRef
                    .current
                    .values()
                )
                  .filter(
                    (
                      annotation:
                        any
                    ) =>
                      annotation
                        .pageIndex ===
                      pageIndex
                  )
                  .reverse();

              for (
                const image
                of images
              ) {
                const box =
                  image
                    .boundingBox;

                if (
                  !box
                ) {
                  continue;
                }

                const insideX =
                  point.x >=
                    box.left &&
                  point.x <=
                    box.left +
                      box.width;

                const insideY =
                  point.y >=
                    box.top &&
                  point.y <=
                    box.top +
                      box.height;

                if (
                  insideX &&
                  insideY
                ) {
                  if (
                    typeof image.id ===
                    "string"
                  ) {
                    imageId =
                      image.id;
                  }

                  break;
                }
              }

              if (
                imageId
              ) {
                break;
              }
            }

            if (
              !imageId
            ) {
              return;
            }

            pointerEvent.preventDefault();

            pointerEvent.stopPropagation();

            pointerEvent.stopImmediatePropagation();

            void selectImageForEditing(
              imageId
            );
          };

        contentDocument.addEventListener(
          "pointerdown",
          handlePointerDown,
          true
        );

        imageHitTestCleanupRef.current =
          () => {
            contentDocument.removeEventListener(
              "pointerdown",
              handlePointerDown,
              true
            );
          };
      }

      /* =====================================================
         REAL RESIZABLE ADD TEXT

         IMPORTANT:

         Existing PDF text:
           -> still uses Content Editor.

         New text added by the user:
           -> uses Nutrient TextAnnotation.

         This gives the new text its own real annotation
         boundingBox with width + height and native
         annotation resizing.

         LOCKED FEATURES NOT TOUCHED:
         - image re-selection
         - image drag / resize
         - Search
         - Unit Converter
         - Save / Download
         - PDF content text editing
         - Locate
         - mouse zoom
      ===================================================== */

      async function startResizableTextAnnotationTool() {
        const instance =
          instanceRef.current;

        const NutrientViewer =
          nutrientRef.current;

        if (
          !instance ||
          !NutrientViewer
        ) {
          return;
        }

        textAnnotationWorkflowRef.current =
          true;

        textAnnotationSelectionArmedRef.current =
          false;

        /*
         * We still want to return to Content Editor
         * after the new annotation has been finished.
         */
        contentEditorWorkflowRef.current =
          true;

        /*
         * Content Editor and TextAnnotation creation
         * cannot be active at the same time.
         *
         * Save the current Content Editor state first.
         */
        try {
          if (
            instance
              .contentEditor
              .isActive()
          ) {
            await instance
              .saveContentEditingSession();
          }
        } catch (
          error
        ) {
          console.error(
            "Unable to temporarily close Content Editor:",
            error
          );

          textAnnotationWorkflowRef.current =
            false;

          return;
        }

        try {
          instance.setEditingAnnotation(
            null
          );
        } catch {
          // Nothing being edited.
        }

        try {
          instance.setSelectedAnnotations(
            null
          );
        } catch {
          // Nothing selected.
        }

        /*
         * Official Nutrient free-text annotation mode.
         *
         * The next click on the PDF creates
         * a TextAnnotation rather than a
         * Content Editor TextBlock.
         */
        instance.setViewState(
          (
            state

            state.set(
              "interactionMode",
              NutrientViewer
                .InteractionMode
                .TEXT

      }


      /* =====================================================
         PREPARE NEW TEXTANNOTATION

         Gives every newly-added TextAnnotation
         a useful initial WIDTH + HEIGHT.
      ===================================================== */

      async function prepareNewTextAnnotation(
        annotation:
          any
      ) {
        const instance =
          instanceRef.current;

        const NutrientViewer =
          nutrientRef.current;

        if (
          !instance ||
          !NutrientViewer ||
          !annotation ||
          !(
            annotation instanceof
            NutrientViewer
              .Annotations
              .TextAnnotation
          )
        ) {
          return;
        }

        const annotationId =
          annotation.id;

        const pageIndex =
          annotation.pageIndex;

        const originalBox =
          annotation.boundingBox;

        if (
          typeof annotationId !==
            "string" ||
          typeof pageIndex !==
            "number" ||
          !originalBox
        ) {
          return;
        }

        const pageInfo =
          instance.pageInfoForIndex(
            pageIndex
          );

        if (
          !pageInfo
        ) {
          return;
        }

        /*
         * Default new text area.
         *
         * User can resize it afterward.
         */
        const desiredWidth =
          260;

        const desiredHeight =
          90;

        const margin =
          10;

        const maxWidth =
          Math.max(
            60,
            pageInfo.width -
              originalBox.left -
              margin
          );

        const maxHeight =
          Math.max(
            35,
            pageInfo.height -
              originalBox.top -
              margin
          );

        const width =
          Math.min(
            maxWidth,
            Math.max(
              originalBox.width,
              desiredWidth
            )
          );

        const height =
          Math.min(
            maxHeight,
            Math.max(
              originalBox.height,
              desiredHeight
            )
          );

        let updatedAnnotation =
          annotation;

        /*
         * Give the annotation a real 2D area.
         */
        updatedAnnotation =
          updatedAnnotation
            .set(
              "boundingBox",
              new NutrientViewer
                .Geometry
                .Rect({
                  left:
                    originalBox.left,

                  top:
                    originalBox.top,

                  width,

                  height,
                })
            )
            .set(
              "isFitting",
              false
            )
            .set(
              "locked",
              false
            )
            .set(
              "lockedContents",
              false
            )
            .set(
              "readOnly",
              false
            );

        try {
          await instance.update(
            updatedAnnotation
          );
        } catch (
          error
        ) {
          console.error(
            "Unable to prepare new TextAnnotation:",
            error
          );

          return;
        }

        /*
         * Leave the repeated-create TEXT tool.
         * We only want one annotation for this click.
         */
        instance.setViewState(
          (
            state
          ) =>
            state.set(
              "interactionMode",
              null
            )
        );

        /*
         * Put the new TextAnnotation directly
         * into normal Nutrient text editing.
         */
        window.requestAnimationFrame(
          () => {
            window.requestAnimationFrame(
              () => {
                try {
                  instance
                    .setEditingAnnotation(
                      annotationId,
                      false
                    );
                } catch (
                  error
                ) {
                  console.error(
                    "Unable to edit new TextAnnotation:",
                    error
                  );
                }

                /*
                 * Arm deselection only AFTER
                 * Nutrient has mounted its
                 * annotation selection.
                 */
                window.requestAnimationFrame(
                  () => {
                    textAnnotationSelectionArmedRef.current =
                      true;
                  }
                );
              }
            );
          }
        );
      }


      /* =====================================================
         RETURN FROM TEXTANNOTATION TO CONTENT EDITOR
      ===================================================== */

      function returnToContentEditorAfterTextAnnotation() {
        textAnnotationWorkflowRef.current =
          false;

        textAnnotationSelectionArmedRef.current =
          false;

        const instance =
          instanceRef.current;

        if (
          !instance
        ) {
          return;
        }

        /*
         * Give Nutrient time to finish the
         * annotation deselection / editing event.
         */
        window.setTimeout(
          () => {
            if (
              contentEditorWorkflowRef.current &&
              !textAnnotationWorkflowRef.current
            ) {
              activateContentEditor();
            }
          },
          100
        );
      }


      /* =====================================================
         TEXTANNOTATION EVENTS
      ===================================================== */

      function installTextAnnotationWorkflow(
        instance:
          NutrientInstance,

        NutrientViewer:
          NutrientViewerType
      ) {
        textAnnotationEventsCleanupRef.current?.();

        /*
         * When the TEXT tool creates a TextAnnotation,
         * enlarge it and open it for typing.
         */
        const createdHandler =
          (
            annotations:
              any
          ) => {
            if (
              !textAnnotationWorkflowRef.current ||
              !annotations
            ) {
              return;
            }

            annotations.forEach(
              (
                annotation:
                  any
              ) => {
                if (
                  annotation instanceof
                  NutrientViewer
                    .Annotations
                    .TextAnnotation
                ) {
                  void prepareNewTextAnnotation(
                    annotation
                  );
                }
              }
            );
          };

        /*
         * Clicking OUTSIDE the new text annotation
         * returns the user to the normal Content Editor.
         */
        const selectionHandler =
          (
            annotations:
              any
          ) => {
            if (
              !textAnnotationWorkflowRef.current ||
              !textAnnotationSelectionArmedRef.current
            ) {
              return;
            }

            if (
              annotations &&
              annotations.size >
                0
            ) {
              return;
            }

            returnToContentEditorAfterTextAnnotation();
          };

        /*
         * If the user deletes the new text annotation,
         * also return normally to Content Editor.
         */
        const deleteHandler =
          (
            annotations:
              any
          ) => {
            if (
              !textAnnotationWorkflowRef.current ||
              !annotations
            ) {
              return;
            }

            let textDeleted =
              false;

            annotations.forEach(
              (
                annotation:
                  any
              ) => {
                if (
                  annotation instanceof
                  NutrientViewer
                    .Annotations
                    .TextAnnotation
                ) {
                  textDeleted =
                    true;
                }
              }
            );

            if (
              textDeleted
            ) {
              returnToContentEditorAfterTextAnnotation();
            }
          };

        instance.addEventListener(
          "annotations.create",
          createdHandler
        );

        instance.addEventListener(
          "annotationSelection.change",
          selectionHandler
        );

        instance.addEventListener(
          "annotations.delete",
          deleteHandler
        );

        textAnnotationEventsCleanupRef.current =
          () => {
            try {
              instance.removeEventListener(
                "annotations.create",
                createdHandler
              );

              instance.removeEventListener(
                "annotationSelection.change",
                selectionHandler
              );

              instance.removeEventListener(
                "annotations.delete",
                deleteHandler
              );
            } catch {
              // Already removed.
            }
          };
      }


      /* =====================================================

        instance:
          NutrientInstance
      ) {
        addTextOverrideCleanupRef.current?.();

        const contentDocument =
          instance.contentDocument;

        let patchFrame =
          0;

        const patchButtons =
          () => {
            cancelAnimationFrame(
              patchFrame
            );

            patchFrame =
              requestAnimationFrame(
                () => {
                  /*
                   * Only patch this button while the
                   * PDF Content Editor is active.
                   */
                  if (
                    !instance
                      .contentEditor
                      .isActive()
                  ) {
                    return;
                  }

                  const buttons =
                    Array.from(
                      contentDocument
                        .querySelectorAll(
                          "button"
                        )
                    ) as
                      HTMLButtonElement[];

                  buttons.forEach(
                    (
                      button
                    ) => {
                      if (
                        button.dataset
                          .pdfRealTextAnnotation ===
                        "true"
                      ) {
                        return;
                      }

                      const label =
                        [
                          button.textContent,
                          button.getAttribute(
                            "title"
                          ),
                          button.getAttribute(
                            "aria-label"
                          ),
                          button.getAttribute(
                            "data-tooltip"
                          ),
                        ]
                          .filter(
                            Boolean
                          )
                          .join(
                            " "
                          )
                          .trim()
                          .toLowerCase();

                      /*
                       * Nutrient currently calls this
                       * Content Editor control
                       * "Add text box".
                       *
                       * We intentionally keep matching
                       * narrow so no other toolbar
                       * feature is touched.
                       */
                      const isAddText =
                        label.includes(
                          "add text box"
                        ) ||
                        label ===
                          "add text" ||
                        label.includes(
                          "add new text"
                        );

                      if (
                        !isAddText
                      ) {
                        return;
                      }

                      /*
                       * Clone preserves the exact
                       * Nutrient icon / styling but
                       * removes Nutrient's old
                       * Content Editor click handler.
                       */
                      const replacement =
                        button.cloneNode(
                          true
                        ) as
                          HTMLButtonElement;

                      replacement.dataset
                        .pdfRealTextAnnotation =
                        "true";

                      replacement.setAttribute(
                        "title",

                      );

                      replacement.setAttribute(
                        "aria-label",

                      );

                      button.replaceWith(
                        replacement
                      );
                    }
                  );
                }
              );
          };

        const observer =
          new MutationObserver(
            patchButtons
          );

        observer.observe(
          contentDocument,
          {
            childList:
              true,

            subtree:
              true,

            attributes:
              true,

            attributeFilter:
              [
                "title",
                "aria-label",
              ],
          }
        );

        const stateChangeHandler =
          () => {
            patchButtons();
          };

        instance.addEventListener(
          "contentEditor.stateChange",
          stateChangeHandler
        );

        patchButtons();

        addTextOverrideCleanupRef.current =
          () => {
            observer.disconnect();

            cancelAnimationFrame(
              patchFrame
            );

            try {
              instance.removeEventListener(
                "contentEditor.stateChange",
                stateChangeHandler
              );
            } catch {
              // Already removed.
            }
          };

      /* =====================================================
         NEW:
         SEARCH RESULT -> BLUE EDITABLE CONTENT BOX
      ===================================================== */

      function rectCenter(
        rect:
          any
      ) {
        return {
          x:
            Number(
              rect?.left ??
              0
            ) +
            Number(
              rect?.width ??
              0
            ) /
              2,

          y:
            Number(
              rect?.top ??
              0
            ) +
            Number(
              rect?.height ??
              0
            ) /
              2,
        };
      }

      function distanceBetweenRects(
        first:
          any,

        second:
          any
      ) {
        if (
          !first ||
          !second
        ) {
          return Number
            .POSITIVE_INFINITY;
        }

        const firstCenter =
          rectCenter(
            first
          );

        const secondCenter =
          rectCenter(
            second
          );

        return (
          Math.abs(
            firstCenter.x -
              secondCenter.x
          ) +
          Math.abs(
            firstCenter.y -
              secondCenter.y
          )
        );
      }

      async function waitForMatchingContentBlock(
        pageIndex:
          number,

        term:
          string,

        resultRect:
          any
      ):
        Promise<any | null> {
        const instance =
          instanceRef.current;

        if (
          !instance
        ) {
          return null;
        }

        const searchTerm =
          term
            .trim()
            .toLocaleLowerCase();

        for (
          let attempt = 0;
          attempt <
            30;
          attempt++
        ) {
          const blocks =
            instance
              .contentEditor
              .getBlocks(
                pageIndex
              ) as
              readonly any[];

          if (
            blocks.length >
            0
          ) {
            const textBlocks =
              blocks.filter(
                (
                  block:
                    any
                ) =>
                  block &&
                  typeof block.id ===
                    "string" &&
                  typeof block.text ===
                    "string"
              );

            const matchingBlocks =
              textBlocks.filter(
                (
                  block:
                    any
                ) =>
                  block.text
                    .toLocaleLowerCase()
                    .includes(
                      searchTerm
                    )
              );

            const candidates =
              matchingBlocks.length >
              0
                ? matchingBlocks
                : textBlocks;

            if (
              candidates.length >
              0
            ) {
              let bestBlock =
                candidates[0];

              let bestDistance =
                Number
                  .POSITIVE_INFINITY;

              for (
                const block
                of candidates
              ) {
                const blockRect =
                  block
                    ?.boundingBox;

                if (
                  !blockRect
                ) {
                  if (
                    bestDistance ===
                    Number
                      .POSITIVE_INFINITY
                  ) {
                    bestBlock =
                      block;
                  }

                  continue;
                }

                const distance =
                  distanceBetweenRects(
                    blockRect,
                    resultRect
                  );

                if (
                  distance <
                  bestDistance
                ) {
                  bestDistance =
                    distance;

                  bestBlock =
                    block;
                }
              }

              return bestBlock;
            }
          }

          await delay(
            80
          );
        }

        return null;
      }

      async function openSearchResultInContentEditor(
        searchResult:
          any,

        term:
          string
      ) {
        const instance =
          instanceRef.current;

        const NutrientViewer =
          nutrientRef.current;

        if (
          !instance ||
          !NutrientViewer ||
          searchResultOpeningRef.current
        ) {
          return;
        }

        const cleanedTerm =
          term.trim();

        if (
          !cleanedTerm
        ) {
          return;
        }

        searchResultOpeningRef.current =
          true;

        try {
          const pageIndex =
            Number(
              searchResult
                ?.pageIndex
            );

          if (
            !Number.isInteger(
              pageIndex
            ) ||
            pageIndex <
              0 ||
            pageIndex >=
              instance
                .totalPageCount
          ) {
            return;
          }

          const rects =
            searchResult
              ?.rectsOnPage;

          if (
            !rects ||
            rects.size ===
              0
          ) {
            return;
          }

          const resultRect =
            NutrientViewer
              .Geometry
              .Rect
              .union(
                rects
              );

          /*
           * Search was started from the Content Editor workflow.
           * Keep that workflow active.
           */
          contentEditorWorkflowRef.current =
            true;

          resumeAfterSearchRef.current =
            false;

          /*
           * Stay at the searched result.
           * Do NOT zoom.
           */
          instance.jumpToRect(
            pageIndex,
            resultRect.grow(
              22
            )
          );

          /*
           * Search mode and Content Editor mode cannot
           * operate simultaneously.
           *
           * We now switch from Search to Content Editor.
           */
          activateContentEditor();

          /*
           * Allow page + blue Content Editor blocks
           * to render/detect.
           */
          await nextFrame();
          await nextFrame();

          const block =
            await waitForMatchingContentBlock(
              pageIndex,
              cleanedTerm,
              resultRect
            );

          if (
            !block ||
            typeof block.id !==
              "string"
          ) {
            /*
             * Content Editor is still open.
             * User can manually click the nearby blue box.
             */
            return;
          }

          /*
           * Put searched block directly into edit mode.
           */
          instance
            .contentEditor
            .focusBlock(
              block.id
            );

          await nextFrame();

          const text =
            typeof block.text ===
            "string"
              ? block.text
              : "";

          const lowerText =
            text.toLocaleLowerCase();

          const lowerTerm =
            cleanedTerm
              .toLocaleLowerCase();

          const matchIndex =
            lowerText.indexOf(
              lowerTerm
            );

          if (
            matchIndex <
            0
          ) {
            /*
             * Block is focused even when exact character
             * mapping is unavailable.
             */
            return;
          }

          /*
           * Select only the searched word inside
           * the blue editable text block.
           */
          await instance
            .contentEditor
            .setTextSelection(
              block.id,
              {
                begin:
                  matchIndex,

                end:
                  matchIndex +
                  cleanedTerm.length,
              }
            );
        } catch (
          error
        ) {
          console.error(
            "Unable to open searched text in Content Editor:",
            error
          );
        } finally {
          searchResultOpeningRef.current =
            false;
        }
      }

      function installSearchToEditBehavior(
        instance:
          NutrientInstance,

        NutrientViewer:
          NutrientViewerType
      ) {
        searchEditCleanupRef.current?.();

        const contentDocument =
          instance
            .contentDocument;

        /*
         * Keep the latest built-in search state.
         */
        const handleSearchStateChange =
          (
            state:
              any
          ) => {
            searchStateRef.current =
              state;
          };

        instance.addEventListener(
          "search.stateChange",
          handleSearchStateChange as any
        );

        searchStateRef.current =
          instance
            .searchState;

        /*
         * User searches normally.
         *
         * Yellow search result remains visible.
         *
         * Clicking that highlighted search result
         * moves it directly into Content Editor.
         */
        const handleSearchResultPointerDown =
          (
            event:
              Event
          ) => {
            const pointerEvent =
              event as
                PointerEvent;

            if (
              pointerEvent.button !==
              0
            ) {
              return;
            }

            if (
              searchResultOpeningRef.current
            ) {
              return;
            }

            if (
              instance
                .viewState
                .interactionMode !==
              NutrientViewer
                .InteractionMode
                .SEARCH
            ) {
              return;
            }

            const searchState =
              searchStateRef.current ??
              instance.searchState;

            const term =
              String(
                searchState
                  ?.term ??
                ""
              ).trim();

            const results =
              searchState
                ?.results;

            if (
              !term ||
              !results ||
              results.size ===
                0
            ) {
              return;
            }

            /*
             * Don't steal Search UI button/input clicks.
             */
            const target =
              pointerEvent.target as
                Element |
                null;

            if (
              target &&
              typeof target.closest ===
                "function" &&
              target.closest(
                [
                  "button",
                  "input",
                  "select",
                  "textarea",
                  "a",
                  "[role='button']",
                ].join(
                  ","
                )
              )
            ) {
              return;
            }

            let hitResult:
              any =
                null;

            results.forEach(
              (
                result:
                  any
              ) => {
                if (
                  hitResult
                ) {
                  return;
                }

                const pageIndex =
                  Number(
                    result
                      ?.pageIndex
                  );

                const rects =
                  result
                    ?.rectsOnPage;

                if (
                  !Number.isInteger(
                    pageIndex
                  ) ||
                  !rects ||
                  rects.size ===
                    0
                ) {
                  return;
                }

                let point:
                  any;

                try {
                  point =
                    instance.transformContentClientToPageSpace(
                      new NutrientViewer
                        .Geometry
                        .Point({
                          x:
                            pointerEvent
                              .clientX,

                          y:
                            pointerEvent
                              .clientY,
                        }),

                      pageIndex
                    );
                } catch {
                  return;
                }

                rects.forEach(
                  (
                    rect:
                      any
                  ) => {
                    if (
                      hitResult
                    ) {
                      return;
                    }

                    /*
                     * Small padding makes clicking the
                     * yellow highlight easier.
                     */
                    const padding =
                      5;

                    const left =
                      rect.left -
                      padding;

                    const top =
                      rect.top -
                      padding;

                    const right =
                      rect.left +
                      rect.width +
                      padding;

                    const bottom =
                      rect.top +
                      rect.height +
                      padding;

                    if (
                      point.x >=
                        left &&
                      point.x <=
                        right &&
                      point.y >=
                        top &&
                      point.y <=
                        bottom
                    ) {
                      hitResult =
                        result;
                    }
                  }
                );
              }
            );

            if (
              !hitResult
            ) {
              return;
            }

            pointerEvent.preventDefault();

            pointerEvent.stopPropagation();

            pointerEvent.stopImmediatePropagation();

            void openSearchResultInContentEditor(
              hitResult,
              term
            );
          };

        contentDocument.addEventListener(
          "pointerdown",
          handleSearchResultPointerDown,
          true
        );

        searchEditCleanupRef.current =
          () => {
            contentDocument.removeEventListener(
              "pointerdown",
              handleSearchResultPointerDown,
              true
            );

            try {
              instance.removeEventListener(
                "search.stateChange",
                handleSearchStateChange as any
              );
            } catch {
              // Already removed.
            }
          };
      }

      /* =====================================================
         INSERT IMAGE
      ===================================================== */

      async function insertImage() {
        const instance =
          instanceRef.current;

        const NutrientViewer =
          nutrientRef.current;

        if (
          !instance ||
          !NutrientViewer ||
          imageInsertBusyRef.current
        ) {
          return;
        }

        const input =
          document.createElement(
            "input"
          );

        input.type =
          "file";

        input.accept =
          "image/png,image/jpeg,image/jpg";

        input.style.display =
          "none";

        document.body.appendChild(
          input
        );

        input.onchange =
          async () => {
            const file =
              input.files?.[0];

            input.remove();

            if (
              !file
            ) {
              return;
            }

            imageInsertBusyRef.current =
              true;

            try {
              await finishContentEditorIfNeeded();

              const imageAttachmentId =
                await instance.createAttachment(
                  file
                );

              const dimensions =
                await getImageDimensions(
                  file
                );

              const pageIndex =
                instance
                  .viewState
                  .currentPageIndex;

              const pageInfo =
                instance.pageInfoForIndex(
                  pageIndex
                );

              if (
                !pageInfo
              ) {
                throw new Error(
                  "Unable to read current PDF page size."
                );
              }

              const maxWidth =
                pageInfo.width *
                0.35;

              const maxHeight =
                pageInfo.height *
                0.35;

              const ratio =
                dimensions.width /
                Math.max(
                  dimensions.height,
                  1
                );

              let width =
                Math.min(
                  maxWidth,
                  Math.max(
                    120,
                    dimensions.width *
                      0.5
                  )
                );

              let height =
                width /
                ratio;

              if (
                height >
                maxHeight
              ) {
                height =
                  maxHeight;

                width =
                  height *
                  ratio;
              }

              width =
                Math.max(
                  80,
                  width
                );

              height =
                Math.max(
                  60,
                  height
                );

              const left =
                Math.max(
                  10,
                  (
                    pageInfo.width -
                    width
                  ) /
                    2
                );

              const top =
                Math.max(
                  10,
                  (
                    pageInfo.height -
                    height
                  ) /
                    2
                );

              const annotationId =
                NutrientViewer
                  .generateInstantId();

              const annotation =
                new NutrientViewer
                  .Annotations
                  .ImageAnnotation({
                    id:
                      annotationId,

                    pageIndex,

                    contentType:
                      file.type ||
                      "image/png",

                    imageAttachmentId,

                    fileName:
                      file.name,

                    boundingBox:
                      new NutrientViewer
                        .Geometry
                        .Rect({
                          left,
                          top,
                          width,
                          height,
                        }),

                    locked:
                      false,

                    lockedContents:
                      false,

                    readOnly:
                      false,
                  });

              await instance.create(
                annotation
              );

              cacheImageAnnotation(
                annotation
              );

              /*
               * LOCKED WORKING BEHAVIOR:
               * inserted image immediately gets handles.
               */
              await selectImageForEditing(
                annotationId
              );
            } catch (
              error
            ) {
              console.error(
                "Image insertion error:",
                error
              );

              alert(
                error instanceof
                  Error
                  ? error.message
                  : "Unable to insert image."
              );

              imageEditingRef.current =
                false;

              imageSelectionConfirmedRef.current =
                false;

              if (
                contentEditorWorkflowRef.current
              ) {
                activateContentEditor();
              }
            } finally {
              imageInsertBusyRef.current =
                false;
            }
          };

        input.oncancel =
          () => {
            input.remove();
          };

        input.click();
      }

      /* =====================================================
         CANCEL -> UNDO
      ===================================================== */

      function installUndoButton(
        instance:
          NutrientInstance
      ) {
        undoCleanupRef.current?.();

        const contentDocument =
          instance.contentDocument;

        let frame =
          0;

        const patch =
          () => {
            cancelAnimationFrame(
              frame
            );

            frame =
              requestAnimationFrame(
                () => {
                  if (
                    !instance
                      .contentEditor
                      .isActive()
                  ) {
                    return;
                  }

                  const buttons =
                    Array.from(
                      contentDocument
                        .querySelectorAll(
                          "button"
                        )
                    ) as HTMLButtonElement[];

                  const existing =
                    buttons.find(
                      (
                        button
                      ) =>
                        button
                          .dataset
                          .pdfUndo ===
                        "true"
                    );

                  const active =
                    instance
                      .contentEditor
                      .getActiveBlock();

                  const selected =
                    instance
                      .contentEditor
                      .getSelectedBlock();

                  const enabled =
                    Boolean(
                      active ||
                      selected
                    );

                  if (
                    existing
                  ) {
                    existing.disabled =
                      !enabled;

                    existing.style.opacity =
                      enabled
                        ? "1"
                        : "0.45";

                    existing.style.cursor =
                      enabled
                        ? "pointer"
                        : "not-allowed";

                    return;
                  }

                  const cancel =
                    buttons.find(
                      (
                        button
                      ) => {
                        const text =
                          button
                            .textContent
                            ?.trim()
                            .toLowerCase();

                        const aria =
                          button
                            .getAttribute(
                              "aria-label"
                            )
                            ?.trim()
                            .toLowerCase();

                        const title =
                          button
                            .getAttribute(
                              "title"
                            )
                            ?.trim()
                            .toLowerCase();

                        return (
                          text ===
                            "cancel" ||
                          aria ===
                            "cancel" ||
                          title ===
                            "cancel"
                        );
                      }
                    );

                  if (
                    !cancel
                  ) {
                    return;
                  }

                  const undo =
                    cancel.cloneNode(
                      true
                    ) as HTMLButtonElement;

                  undo.dataset.pdfUndo =
                    "true";

                  undo.textContent =
                    "Undo";

                  undo.setAttribute(
                    "title",
                    "Undo"
                  );

                  undo.setAttribute(
                    "aria-label",
                    "Undo"
                  );

                  undo.disabled =
                    !enabled;

                  undo.addEventListener(
                    "click",
                    (
                      event
                    ) => {
                      event.preventDefault();

                      event.stopPropagation();

                      event.stopImmediatePropagation();

                      try {
                        let block =
                          instance
                            .contentEditor
                            .getActiveBlock();

                        if (
                          !block
                        ) {
                          block =
                            instance
                              .contentEditor
                              .getSelectedBlock();

                          if (
                            block
                          ) {
                            instance
                              .contentEditor
                              .focusBlock(
                                block
                              );
                          }
                        }

                        requestAnimationFrame(
                          () => {
                            try {
                              if (
                                instance
                                  .contentEditor
                                  .canUndo()
                              ) {
                                instance
                                  .contentEditor
                                  .undo();
                              }
                            } catch (
                              error
                            ) {
                              console.error(
                                "Undo error:",
                                error
                              );
                            }

                            setTimeout(
                              patch,
                              0
                            );
                          }
                        );
                      } catch (
                        error
                      ) {
                        console.error(
                          "Undo error:",
                          error
                        );
                      }
                    },
                    true
                  );

                  cancel.replaceWith(
                    undo
                  );
                }
              );
          };

        const observer =
          new MutationObserver(
            patch
          );

        observer.observe(
          contentDocument,
          {
            childList:
              true,

            subtree:
              true,
          }
        );

        const stateChanged =
          () => {
            patch();
          };

        instance.addEventListener(
          "contentEditor.stateChange",
          stateChanged
        );

        patch();

        undoCleanupRef.current =
          () => {
            observer.disconnect();

            cancelAnimationFrame(
              frame
            );

            try {
              instance.removeEventListener(
                "contentEditor.stateChange",
                stateChanged
              );
            } catch {
              // Already removed.
            }
          };
      }

      /* =====================================================
         MOUSE PAN + MOUSE WHEEL ZOOM
      ===================================================== */

      function installMouseControls(
        instance:
          NutrientInstance,

        NutrientViewer:
          NutrientViewerType,

        attempt =
          0
      ) {
        mouseCleanupRef.current?.();

        const contentDocument =
          instance.contentDocument;

        const contentWindow =
          instance.contentWindow;

        const scroll =
          contentDocument.querySelector(
            ".PSPDFKit-Scroll"
          ) as HTMLElement | null;

        if (
          !scroll
        ) {
          if (
            attempt <
            12
          ) {
            mouseInstallTimerRef.current =
              window.setTimeout(
                () => {
                  installMouseControls(
                    instance,
                    NutrientViewer,
                    attempt +
                      1
                  );
                },
                150
              );
          }

          return;
        }

        let dragging =
          false;

        let startX =
          0;

        let startY =
          0;

        let startLeft =
          0;

        let startTop =
          0;

        let wheelBusy =
          false;

        const isPanMode =
          () =>
            instance
              .viewState
              .interactionMode ===
            NutrientViewer
              .InteractionMode
              .PAN;

        const isInteractive =
          (
            target:
              EventTarget |
              null
          ) => {
            if (
              !target
            ) {
              return false;
            }

            const element =
              target as
                Element;

            if (
              typeof element.closest !==
              "function"
            ) {
              return false;
            }

            return Boolean(
              element.closest(
                [
                  "button",
                  "input",
                  "select",
                  "textarea",
                  "a",
                  "[role='button']",
                  "[contenteditable='true']",
                ].join(
                  ","
                )
              )
            );
          };

        const mouseDown =
          (
            event:
              MouseEvent
          ) => {
            /*
             * IMPORTANT:
             * never steal mouse events from selected images.
             */
            if (
              imageEditingRef.current
            ) {
              return;
            }

            if (
              event.button !==
                0 ||
              !isPanMode() ||
              isInteractive(
                event.target
              )
            ) {
              return;
            }

            dragging =
              true;

            startX =
              event.clientX;

            startY =
              event.clientY;

            startLeft =
              scroll.scrollLeft;

            startTop =
              scroll.scrollTop;

            scroll.style.cursor =
              "grabbing";

            scroll.style.userSelect =
              "none";

            clearLocateHighlight();

            event.preventDefault();
          };

        const mouseMove =
          (
            event:
              MouseEvent
          ) => {
            if (
              !dragging
            ) {
              return;
            }

            scroll.scrollLeft =
              startLeft -
              (
                event.clientX -
                startX
              );

            scroll.scrollTop =
              startTop -
              (
                event.clientY -
                startY
              );

            event.preventDefault();
          };

        const mouseUp =
          () => {
            if (
              !dragging
            ) {
              return;
            }

            dragging =
              false;

            scroll.style.cursor =
              isPanMode()
                ? "grab"
                : "";

            scroll.style.userSelect =
              "";
          };

        const wheel =
          (
            event:
              WheelEvent
          ) => {
            event.preventDefault();

            event.stopPropagation();

            if (
              wheelBusy
            ) {
              return;
            }

            wheelBusy =
              true;

            requestAnimationFrame(
              () => {
                try {
                  const current =
                    instance
                      .currentZoomLevel;

                  const min =
                    instance
                      .minimumZoomLevel;

                  const max =
                    instance
                      .maximumZoomLevel;

                  let next =
                    event.deltaY <
                    0
                      ? current *
                        1.12
                      : current /
                        1.12;

                  next =
                    Math.max(
                      min,
                      Math.min(
                        max,
                        next
                      )
                    );

                  instance.setViewState(
                    (
                      state
                    ) =>
                      state.set(
                        "zoom",
                        next
                      )
                  );
                } finally {
                  wheelBusy =
                    false;
                }
              }
            );
          };

        const viewChanged =
          () => {
            const currentMode =
              instance
                .viewState
                .interactionMode;

            const previousMode =
              previousInteractionModeRef.current;

            if (
              resumeAfterSearchRef.current &&
              previousMode ===
                NutrientViewer
                  .InteractionMode
                  .SEARCH &&
              currentMode !==
                NutrientViewer
                  .InteractionMode
                  .SEARCH
            ) {
              resumeAfterSearchRef.current =
                false;

              setTimeout(
                () => {
                  if (
                    contentEditorWorkflowRef.current &&
                    currentMode !==
                      NutrientViewer
                        .InteractionMode
                        .CONTENT_EDITOR
                  ) {
                    activateContentEditor();
                  }
                },
                80
              );
            }

            previousInteractionModeRef.current =
              currentMode;
          };

        previousInteractionModeRef.current =
          instance
            .viewState
            .interactionMode;

        scroll.addEventListener(
          "mousedown",
          mouseDown,
          true
        );

        contentWindow.addEventListener(
          "mousemove",
          mouseMove,
          true
        );

        contentWindow.addEventListener(
          "mouseup",
          mouseUp,
          true
        );

        contentWindow.addEventListener(
          "blur",
          mouseUp
        );

        scroll.addEventListener(
          "wheel",
          wheel,
          {
            passive:
              false,

            capture:
              true,
          }
        );

        instance.addEventListener(
          "viewState.change",
          viewChanged
        );

        mouseCleanupRef.current =
          () => {
            scroll.removeEventListener(
              "mousedown",
              mouseDown,
              true
            );

            contentWindow.removeEventListener(
              "mousemove",
              mouseMove,
              true
            );

            contentWindow.removeEventListener(
              "mouseup",
              mouseUp,
              true
            );

            contentWindow.removeEventListener(
              "blur",
              mouseUp
            );

            scroll.removeEventListener(
              "wheel",
              wheel,
              true
            );

            try {
              instance.removeEventListener(
                "viewState.change",
                viewChanged
              );
            } catch {
              // Already removed.
            }
          };
      }

          async analyzeDimensions(
            options:
              AnalyzeDimensionsOptions
          ) {
            const instance =
              instanceRef.current;

            if (
              !instance
            ) {
              throw new Error(
                "PDF editor is not ready."
              );
            }

            if (
              options.sourceUnit ===
              options.targetUnit
            ) {
              throw new Error(
                "Source and target units must be different."
              );
            }

            /*
             * =================================================
             * IMPORTANT
             *
             * Unit Converter uses Nutrient's HEADLESS
             * Content Editing session.
             *
             * We are NOT opening or modifying the visible
             * Content Editor workflow here.
             *
             * This protects all currently locked Content
             * Editor behavior.
             * =================================================
             */

            if (
              instance
                .contentEditor
                .isActive()
            ) {
              /*
               * Normally Unit Converter has already left
               * Content Editor mode before Analyze is used.
               *
               * This is just a safety fallback.
               */
              await instance
                .saveContentEditingSession();

              await new Promise<void>(
                (
                  resolve
                ) => {
                  window.setTimeout(
                    resolve,
                    80
                  );
                }
              );
            }


            const totalPages =
              instance
                .totalPageCount;


            const currentPageIndex =
              instance
                .viewState
                .currentPageIndex;


            const pages =
              options.scope ===
              "all"
                ? Array.from(
                    {
                      length:
                        totalPages,
                    },
                    (
                      _,
                      index
                    ) =>
                      index
                  )
                : [
                    currentPageIndex,
                  ];


            /*
             * =================================================
             * STEP 1
             *
             * Detect dimensions exactly as before.
             *
             * This keeps the existing side-panel results
             * completely unchanged.
             * =================================================
             */

            const dimensions:
              DimensionCandidate[] =
                [];


            for (
              const pageIndex
              of pages
            ) {
              const textLines =
                await instance
                  .textLinesForPageIndex(
                    pageIndex
                  );


              textLines.forEach(
                (
                  textLine,
                  lineIndex
                ) => {
                  const box =
                    textLine
                      .boundingBox
                      .toJS();


                  const found =
                    detectDimensionsInLine(
                      textLine.contents,

                      pageIndex,

                      lineIndex,

                      options.sourceUnit,

                      options.targetUnit,

                      {
                        left:
                          box.left,

                        top:
                          box.top,

                        width:
                          box.width,

                        height:
                          box.height,
                      }
                    );


                  dimensions.push(
                    ...found
                  );
                }
              );
            }


            /*
             * No dimensions found.
             *
             * Return the exact same response shape
             * expected by the Unit Converter panel.
             */

            if (
              dimensions.length ===
              0
            ) {
              return {
                dimensions,

                pagesAnalyzed:
                  pages.length,

                totalPages,
              };
            }


            /*
             * =================================================
             * HELPERS
             * =================================================
             */


            function getRectCenter(
              rect:
                {
                  left: number;
                  top: number;
                  width: number;
                  height: number;
                }
            ) {
              return {
                x:
                  rect.left +
                  rect.width /
                    2,

                y:
                  rect.top +
                  rect.height /
                    2,
              };
            }


            /*
             * Measures how close a Nutrient text block
             * is to the detected engineering dimension.
             *
             * Same-page + text match comes first.
             * Geometry decides which duplicate value
             * belongs to which location.
             */

            function blockDistance(
              blockBox:
                {
                  left: number;
                  top: number;
                  width: number;
                  height: number;
                },

              detectedBox:
                PdfRectData
            ) {
              const blockCenter =
                getRectCenter(
                  blockBox
                );


              const detectedCenter =
                getRectCenter(
                  detectedBox
                );


              const horizontalDistance =
                Math.abs(
                  blockCenter.x -
                    detectedCenter.x
                );


              const verticalDistance =
                Math.abs(
                  blockCenter.y -
                    detectedCenter.y
                );


              /*
               * Vertical position matters more
               * for engineering drawing dimensions.
               */

              let score =
                horizontalDistance +
                verticalDistance *
                  3;


              const overlapLeft =
                Math.max(
                  blockBox.left,
                  detectedBox.left
                );


              const overlapTop =
                Math.max(
                  blockBox.top,
                  detectedBox.top
                );


              const overlapRight =
                Math.min(
                  blockBox.left +
                    blockBox.width,

                  detectedBox.left +
                    detectedBox.width
                );


              const overlapBottom =
                Math.min(
                  blockBox.top +
                    blockBox.height,

                  detectedBox.top +
                    detectedBox.height
                );


              const overlaps =
                overlapRight >
                  overlapLeft &&
                overlapBottom >
                  overlapTop;


              /*
               * Strongly prefer a block that actually
               * overlaps the detected PDF text line.
               */

              if (
                overlaps
              ) {
                score -=
                  10000;
              }


              return score;
            }


            /*
             * Convert detected text into a flexible
             * regular expression.
             *
             * Example:
             *
             * Ø .422
             *
             * will also match:
             *
             * Ø.422
             *
             * This handles minor PDF spacing differences
             * while still requiring the same actual text.
             */

            function makeDimensionPattern(
              text:
                string
            ) {
              const escaped =
                text
                  .replace(
                    /[.*+?^${}()|[\]\\]/g,
                    "\\$&"
                  )
                  .replace(
                    /\s+/g,
                    "\\s*"
                  );


              return new RegExp(
                escaped,
                "g"
              );
            }


            /*
             * =================================================
             * STEP 2
             *
             * Start a PROGRAMMATIC Nutrient Content
             * Editing session.
             *
             * This does NOT open the Content Editor UI.
             * =================================================
             */

            let editingSession:
              any =
                null;


            try {
              editingSession =
                await instance
                  .beginContentEditingSession();


              /*
               * All block changes collected here.
               */

              const updatedBlocks:
                Array<{
                  id: string;
                  text: string;
                }> =
                  [];


              let totalRealReplacements =
                0;


              /*
               * =================================================
               * STEP 3
               *
               * Match detected dimensions to real PDF
               * Content Editor text blocks.
               * =================================================
               */

              for (
                const pageIndex
                of pages
              ) {
                const pageDimensions =
                  dimensions.filter(
                    (
                      dimension
                    ) =>
                      dimension
                        .pageIndex ===
                      pageIndex
                  );


                if (
                  pageDimensions.length ===
                  0
                ) {
                  continue;
                }


                /*
                 * Nutrient performs actual PDF content
                 * detection here.
                 */

                const textBlocks =
                  await editingSession
                    .getTextBlocks(
                      pageIndex
                    );


                if (
                  !textBlocks ||
                  textBlocks.length ===
                    0
                ) {
                  console.warn(
                    `No editable PDF text blocks found on page ${pageIndex + 1}.`
                  );

                  continue;
                }


                /*
                 * Store replacements PER original block.
                 *
                 * We don't modify the text immediately.
                 * Multiple dimensions can exist inside
                 * one PDF text block.
                 */

                const replacementsByBlock =
                  new Map<
                    string,
                    Array<{
                      start: number;
                      length: number;
                      replacement: string;
                      original: string;
                    }>
                  >();


                /*
                 * Prevent the same text occurrence from
                 * being claimed by multiple detections.
                 */

                const usedOccurrences =
                  new Set<
                    string
                  >();


                for (
                  const dimension
                  of pageDimensions
                ) {
                  let bestBlock:
                    any =
                      null;


                  let bestStart =
                    -1;


                  let bestLength =
                    0;


                  let bestScore =
                    Number
                      .POSITIVE_INFINITY;


                  const dimensionPattern =
                    makeDimensionPattern(
                      dimension.original
                    );


                  /*
                   * Check every editable block on
                   * the same PDF page.
                   */

                  for (
                    const block
                    of textBlocks
                  ) {
                    if (
                      !block ||
                      typeof block.id !==
                        "string" ||
                      typeof block.text !==
                        "string" ||
                      !block.boundingBox
                    ) {
                      continue;
                    }


                    /*
                     * Fresh RegExp for every block.
                     */

                    const matcher =
                      new RegExp(
                        dimensionPattern.source,
                        "g"
                      );


                    let match:
                      RegExpExecArray |
                      null;


                    while (
                      (
                        match =
                          matcher.exec(
                            block.text
                          )
                      ) !==
                      null
                    ) {
                      const occurrenceKey =
                        `${block.id}:${match.index}:${match[0].length}`;


                      if (
                        usedOccurrences.has(
                          occurrenceKey
                        )
                      ) {
                        continue;
                      }


                      const score =
                        blockDistance(
                          {
                            left:
                              Number(
                                block
                                  .boundingBox
                                  .left
                              ),

                            top:
                              Number(
                                block
                                  .boundingBox
                                  .top
                              ),

                            width:
                              Number(
                                block
                                  .boundingBox
                                  .width
                              ),

                            height:
                              Number(
                                block
                                  .boundingBox
                                  .height
                              ),
                          },

                          dimension
                            .lineBox
                        );


                      if (
                        score <
                        bestScore
                      ) {
                        bestScore =
                          score;

                        bestBlock =
                          block;

                        bestStart =
                          match.index;

                        bestLength =
                          match[0]
                            .length;
                      }


                      /*
                       * Safety against a zero-length regex.
                       */

                      if (
                        match[0].length ===
                        0
                      ) {
                        matcher.lastIndex++;
                      }
                    }
                  }


                  /*
                   * Nothing editable matched this
                   * detected side-panel result.
                   *
                   * Keep it in the side panel exactly
                   * as before, but do NOT replace some
                   * unrelated PDF text.
                   */

                  if (
                    !bestBlock ||
                    bestStart <
                      0 ||
                    bestLength <=
                      0
                  ) {
                    console.warn(
                      "Unit Converter: could not safely match this detected dimension to editable PDF content:",

                      {
                        page:
                          dimension
                            .pageNumber,

                        original:
                          dimension
                            .original,

                        converted:
                          dimension
                            .converted,
                      }
                    );

                    continue;
                  }


                  const occurrenceKey =
                    `${bestBlock.id}:${bestStart}:${bestLength}`;


                  usedOccurrences.add(
                    occurrenceKey
                  );


                  const existing =
                    replacementsByBlock.get(
                      bestBlock.id
                    ) ??
                    [];


                  /*
                   * Real replacement:
                   *
                   * .002
                   *
                   * becomes
                   *
                   * 0.0508 mm
                   *
                   * directly inside the PDF content.
                   */

                  existing.push({
                    start:
                      bestStart,

                    length:
                      bestLength,

                    replacement:
                      dimension
                        .converted,

                    original:
                      dimension
                        .original,
                  });


                  replacementsByBlock.set(
                    bestBlock.id,
                    existing
                  );
                }


                /*
                 * =================================================
                 * STEP 4
                 *
                 * Build updated text for each real
                 * Nutrient PDF text block.
                 * =================================================
                 */

                for (
                  const block
                  of textBlocks
                ) {
                  if (
                    !block ||
                    typeof block.id !==
                      "string" ||
                    typeof block.text !==
                      "string"
                  ) {
                    continue;
                  }


                  const replacements =
                    replacementsByBlock.get(
                      block.id
                    );


                  if (
                    !replacements ||
                    replacements.length ===
                      0
                  ) {
                    continue;
                  }


                  /*
                   * IMPORTANT:
                   *
                   * Replace from RIGHT to LEFT.
                   *
                   * This prevents longer converted text
                   * from shifting the indexes of earlier
                   * dimensions in the same PDF block.
                   */

                  const ordered =
                    [
                      ...replacements,
                    ].sort(
                      (
                        first,
                        second
                      ) =>
                        second.start -
                        first.start
                    );


                  let updatedText =
                    block.text;


                  for (
                    const replacement
                    of ordered
                  ) {
                    const before =
                      updatedText.slice(
                        0,
                        replacement.start
                      );


                    const after =
                      updatedText.slice(
                        replacement.start +
                          replacement.length
                      );


                    updatedText =
                      before +
                      replacement.replacement +
                      after;


                    totalRealReplacements++;
                  }


                  if (
                    updatedText !==
                    block.text
                  ) {
                    updatedBlocks.push({
                      id:
                        block.id,

                      text:
                        updatedText,
                    });
                  }
                }
              }


              /*
               * =================================================
               * STEP 5
               *
               * Write the actual changed text blocks
               * back into Nutrient's PDF content session.
               * =================================================
               */

              if (
                updatedBlocks.length >
                0
              ) {
                await editingSession
                  .updateTextBlocks(
                    updatedBlocks
                  );
              }


              /*
               * =================================================
               * STEP 6
               *
               * Commit to the CURRENT OPEN DOCUMENT.
               *
               * IMPORTANT:
               *
               * This does NOT POST the PDF to your FastAPI
               * backend.
               *
               * The drawing in the browser changes now.
               *
               * Your existing main Save button is still the
               * only place that stores the edited PDF in your
               * backend.
               * =================================================
               */

              if (
                updatedBlocks.length >
                0
              ) {
                const saveResult =
                  await editingSession
                    .commit();


                /*
                 * Font safety warning only.
                 */

                if (
                  saveResult &&
                  saveResult.status ===
                    "reported" &&
                  Array.isArray(
                    saveResult
                      .uncoveredCodepoints
                  ) &&
                  saveResult
                    .uncoveredCodepoints
                    .length >
                    0
                ) {
                  console.warn(
                    "Unit Converter: some converted characters are not available in the original PDF fonts.",

                    saveResult
                      .uncoveredCodepoints
                  );
                }


                console.log(
                  `Unit Converter: ${totalRealReplacements} real PDF dimension replacement(s) applied.`
                );
              }
              else {
                /*
                 * Nothing safely matched.
                 *
                 * End the programmatic session
                 * without changing the PDF.
                 */

                if (
                  editingSession
                    ?.active
                ) {
                  await editingSession
                    .discard();
                }


                console.warn(
                  "Unit Converter: dimensions were detected, but no editable PDF text could be safely replaced."
                );
              }
            }
            catch (
              replacementError
            ) {
              console.error(
                "Unit Converter real PDF replacement error:",
                replacementError
              );


              /*
               * Never leave Nutrient stuck inside
               * a failed programmatic editing session.
               */

              try {
                if (
                  editingSession
                    ?.active
                ) {
                  await editingSession
                    .discard();
                }
              }
              catch (
                discardError
              ) {
                console.error(
                  "Unable to discard failed Unit Converter session:",
                  discardError
                );
              }


              throw new Error(
                replacementError instanceof
                  Error
                  ? `Drawing analysis succeeded, but real PDF text replacement failed: ${replacementError.message}`
                  : "Drawing analysis succeeded, but real PDF text replacement failed."
              );
            }


            /*
             * =================================================
             * SIDE PANEL
             *
             * EXACT SAME RETURN STRUCTURE AS BEFORE.
             *
             * No Unit Converter side-panel UI changes.
             * =================================================
             */

            return {
              dimensions,

              pagesAnalyzed:
                pages.length,

              totalPages,
            };
          },

          async locateDimension(
            dimension
          ) {
            const instance =
              instanceRef.current;

            const NutrientViewer =
              nutrientRef.current;

            if (
              !instance ||
              !NutrientViewer
            ) {
              throw new Error(
                "PDF editor is not ready."
              );
            }

            clearLocateHighlight();

            let targetRect =
              new NutrientViewer
                .Geometry
                .Rect({
                  left:
                    dimension
                      .lineBox
                      .left,

                  top:
                    dimension
                      .lineBox
                      .top,

                  width:
                    dimension
                      .lineBox
                      .width,

                  height:
                    dimension
                      .lineBox
                      .height,
                });

            try {
              const results =
                await instance.search(
                  dimension.original,
                  {
                    startPageIndex:
                      dimension
                        .pageIndex,

                    endPageIndex:
                      dimension
                        .pageIndex,
                  }
                );

              const targetX =
                targetRect.left +
                targetRect.width /
                  2;

              const targetY =
                targetRect.top +
                targetRect.height /
                  2;

              let best =
                Number
                  .POSITIVE_INFINITY;

              results.forEach(
                (
                  result
                ) => {
                  if (
                    result.pageIndex !==
                      dimension
                        .pageIndex ||
                    result
                      .rectsOnPage
                      .size ===
                      0
                  ) {
                    return;
                  }

                  const candidate =
                    NutrientViewer
                      .Geometry
                      .Rect
                      .union(
                        result
                          .rectsOnPage
                      );

                  const x =
                    candidate.left +
                    candidate.width /
                      2;

                  const y =
                    candidate.top +
                    candidate.height /
                      2;

                  const distance =
                    Math.abs(
                      x -
                        targetX
                    ) +
                    Math.abs(
                      y -
                        targetY
                    ) *
                      3;

                  if (
                    distance <
                    best
                  ) {
                    best =
                      distance;

                    targetRect =
                      candidate;
                  }
                }
              );
            } catch {
              // Use original line rect.
            }

            instance.jumpToRect(
              dimension.pageIndex,
              targetRect.grow(
                18
              )
            );

            activatePanMode();

            const highlight =
              document.createElement(
                "div"
              );

            highlight.style.width =
              `${Math.max(
                targetRect.width,
                8
              )}px`;

            highlight.style.height =
              `${Math.max(
                targetRect.height,
                8
              )}px`;

            highlight.style.background =
              "rgba(37,99,235,0.28)";

            highlight.style.border =
              "2px solid #2563eb";

            highlight.style.borderRadius =
              "3px";

            highlight.style.boxSizing =
              "border-box";

            highlight.style.boxShadow =
              "0 0 0 3px rgba(37,99,235,0.16)";

            highlight.style.pointerEvents =
              "none";

            instance.setCustomOverlayItem(
              new NutrientViewer
                .CustomOverlayItem({
                  id:
                    LOCATE_OVERLAY_ID,

                  node:
                    highlight,

                  pageIndex:
                    dimension
                      .pageIndex,

                  position:
                    new NutrientViewer
                      .Geometry
                      .Point({
                        x:
                          targetRect.left,

                        y:
                          targetRect.top,
                      }),
                })
            );
          },

          async saveToBackend() {
            const instance =
              instanceRef.current;

            if (
              !instance
            ) {
              throw new Error(
                "PDF editor is not ready."
              );
            }

            const pdfBuffer =
              await exportWithoutClosingEditor();

            const blob =
              new Blob(
                [
                  pdfBuffer,
                ],
                {
                  type:
                    "application/pdf",
                }
              );

            const formData =
              new FormData();

            formData.append(
              "file",
              blob,
              `${documentId}.pdf`
            );

            const response =
              await fetch(
                `http://127.0.0.1:8000/api/documents/${documentId}/save`,
                {
                  method:
                    "POST",

                  body:
                    formData,
                }
              );

            if (
              !response.ok
            ) {
              let message =
                "Unable to save edited PDF.";

              try {
                const data =
                  await response.json();

                if (
                  typeof data
                    ?.detail ===
                  "string"
                ) {
                  message =
                    data.detail;
                }
              } catch {
                // Keep default.
              }

              throw new Error(
                message
              );
            }

            let warning:
              string |
              undefined;

            if (
              instance
                .contentEditor
                .isActive()
            ) {
              const result =
                await instance
                  .saveContentEditingSession();

              if (
                result.status ===
                  "reported" &&
                result
                  .uncoveredCodepoints
                  .length >
                  0
              ) {
                warning =
                  "Some edited characters may not be supported by the original PDF fonts.";
              }

              if (
                result.status ===
                "reportUnavailable"
              ) {
                warning =
                  "PDF saved, but font support could not be fully verified.";
              }
            }

            contentEditorWorkflowRef.current =
              false;

            imageEditingRef.current =
              false;

            imageSelectionConfirmedRef.current =
              false;

            searchResultOpeningRef.current =
              false;

            try {
              instance.setSelectedAnnotations(
                null
              );
            } catch {
              // Nothing selected.
            }

            activatePanMode();

            return {
              warning,
            };
          },

          async downloadPdf(
            filename
          ) {
            const pdfBuffer =
              await exportWithoutClosingEditor();

            const blob =
              new Blob(
                [
                  pdfBuffer,
                ],
                {
                  type:
                    "application/pdf",
                }
              );

            const url =
              URL.createObjectURL(
                blob
              );

            const cleanName =
              (
                filename ||
                "document.pdf"
              ).replace(
                /\.pdf$/i,
                ""
              );

            const link =
              document.createElement(
                "a"
              );

            link.href =
              url;

            link.download =
              `${cleanName}-edited.pdf`;

            document.body.appendChild(
              link
            );

            link.click();

            link.remove();

            setTimeout(
              () => {
                URL.revokeObjectURL(
                  url
                );
              },
              1000
            );

            return {};
          },
        })
      );

      /* =====================================================
         LOAD VIEWER
      ===================================================== */

      useEffect(
        () => {
          if (
            !documentId
          ) {
            return;
          }

          let cancelled =
            false;

          let loadedInstance:
            NutrientInstance |
            null =
              null;

          let selectionHandler:
            ((annotations: any) => void) |
            null =
              null;

          const controller =
            new AbortController();

          const load =
            async () => {
              const container =
                viewerRef.current;

              if (
                !container
              ) {
                return;
              }

              try {
                onLoadingChangeRef
                  .current?.(
                    true
                  );

                onReadyRef
                  .current?.(
                    false
                  );

                onErrorRef
                  .current?.(
                    ""
                  );

                const key =
                  process.env
                    .NEXT_PUBLIC_NUTRIENT_KEY;

                if (
                  !key
                ) {
                  throw new Error(
                    "NEXT_PUBLIC_NUTRIENT_KEY is missing."
                  );
                }

                const response =
                  await fetch(
                    `http://127.0.0.1:8000/api/documents/${documentId}/file`,
                    {
                      signal:
                        controller
                          .signal,
                    }
                  );

                if (
                  !response.ok
                ) {
                  throw new Error(
                    `Unable to load PDF. HTTP ${response.status}`
                  );
                }

                const buffer =
                  await response
                    .arrayBuffer();

                const module =
                  await import(
                    "@nutrient-sdk/viewer"
                  );

                const NutrientViewer =
                  module.default;

                nutrientRef.current =
                  NutrientViewer;

                await waitForViewerContainer(
                  container,
                  controller.signal
                );

                if (
                  cancelled
                ) {
                  return;
                }

                try {
                  NutrientViewer.unload(
                    container
                  );
                } catch {
                  // Nothing loaded.
                }

                const instance =
                  await NutrientViewer.load({
                    container,

                    baseUrl:
                      `${window.location.origin}/`,

                    session:
                      key,

                    document:
                      buffer,

                    initialViewState:
                      new NutrientViewer
                        .ViewState({
                          showToolbar:
                            true,

                          interactionMode:
                            NutrientViewer
                              .InteractionMode
                              .PAN,
                        }),

                    signal:
                      controller
                        .signal,
                  });

                if (
                  cancelled
                ) {
                  return;
                }

                loadedInstance =
                  instance;

                instanceRef.current =
                  instance;

                applyMainToolbar(
                  instance
                );

                removeNoteOption(
                  instance
                );

                installUndoButton(
                  instance
                );

                /*
                 * TextAnnotation only:
                 * adds top + bottom Y-axis resize handles.
                 */

                /*
                 * LOCKED image behavior.
                 */
                installImageAnnotationEvents(
                  instance,
                  NutrientViewer
                );

                installImageHitTesting(
                  instance,
                  NutrientViewer
                );

                /*
                 * NEW search -> editable blue box behavior.
                 */
                installSearchToEditBehavior(
                  instance,
                  NutrientViewer
                );

                await hydrateImageCacheForPage(
                  instance
                    .viewState
                    .currentPageIndex
                );

                activatePanMode();

                /*
                 * LOCKED image deselection behavior.
                 */
                selectionHandler =
                  (
                    annotations:
                      any
                  ) => {
                    if (
                      !imageEditingRef.current
                    ) {
                      return;
                    }

                    if (
                      annotations &&
                      annotations.size >
                        0
                    ) {
                      imageSelectionConfirmedRef.current =
                        true;

                      return;
                    }

                    if (
                      !imageSelectionConfirmedRef.current
                    ) {
                      return;
                    }

                    returnToContentEditorAfterImage();
                  };

                instance.addEventListener(
                  "annotationSelection.change",
                  selectionHandler
                );

                mouseInstallTimerRef.current =
                  window.setTimeout(
                    () => {
                      if (
                        !cancelled
                      ) {
                        installMouseControls(
                          instance,
                          NutrientViewer
                        );
                      }
                    },
                    500
                  );

                onReadyRef
                  .current?.(
                    true
                  );

                onLoadingChangeRef
                  .current?.(
                    false
                  );
              } catch (
                error
              ) {
                if (
                  cancelled ||
                  controller
                    .signal
                    .aborted
                ) {
                  return;
                }

                console.error(
                  "Nutrient load error:",
                  error
                );

                onErrorRef
                  .current?.(
                    error instanceof
                      Error
                      ? error.message
                      : "Unable to load PDF."
                  );

                onReadyRef
                  .current?.(
                    false
                  );

                onLoadingChangeRef
                  .current?.(
                    false
                  );
              }
            };

          void load();

          return () => {
            cancelled =
              true;

            controller.abort();

            if (
              mouseInstallTimerRef.current !==
              null
            ) {
              clearTimeout(
                mouseInstallTimerRef.current
              );

              mouseInstallTimerRef.current =
                null;
            }

            mouseCleanupRef.current?.();

            mouseCleanupRef.current =
              null;

            imageHitTestCleanupRef.current?.();

            imageHitTestCleanupRef.current =
              null;

            imageEventsCleanupRef.current?.();

            imageEventsCleanupRef.current =
              null;

            searchEditCleanupRef.current?.();

            searchEditCleanupRef.current =
              null;

            undoCleanupRef.current?.();

            undoCleanupRef.current =
              null;

            imageAnnotationsRef.current.clear();

            searchStateRef.current =
              null;

            imageEditingRef.current =
              false;

            imageSelectionConfirmedRef.current =
              false;

            contentEditorWorkflowRef.current =
              false;

            searchResultOpeningRef.current =
              false;

            resumeAfterSearchRef.current =
              false;

            if (
              loadedInstance &&
              selectionHandler
            ) {
              try {
                loadedInstance.removeEventListener(
                  "annotationSelection.change",
                  selectionHandler
                );
              } catch {
                // Already removed.
              }
            }

            if (
              loadedInstance
            ) {
              try {
                loadedInstance.removeCustomOverlayItem(
                  LOCATE_OVERLAY_ID
                );
              } catch {
                // Nothing.
              }
            }

            const container =
              viewerRef.current;

            const NutrientViewer =
              nutrientRef.current;

            if (
              container &&
              NutrientViewer
            ) {
              try {
                NutrientViewer.unload(
                  container
                );
              } catch {
                // Nothing.
              }
            }

            instanceRef.current =
              null;

            nutrientRef.current =
              null;
          };
        },
        [
          documentId,
        ]
      );

      return (
        <div
          ref={
            viewerRef
          }
          className="h-full w-full"
          style={{
            width:
              "100%",

            height:
              "100%",

            minWidth:
              "1px",

            minHeight:
              "420px",
          }}
        />
      );
    }
  );

const NutrientEditor =
  memo(
    NutrientEditorComponent
  );

NutrientEditor.displayName =
  "NutrientEditor";

export default NutrientEditor;
